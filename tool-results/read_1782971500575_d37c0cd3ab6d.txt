   270→   270→  loadData(categories: GraphCategory[], terms: GraphNode['term'][]) {
   271→   271→    this.nodes = []
   272→   272→    this.edges = []
   273→   273→    this.nodeMap.clear()
   274→   274→    this._labelCache.clear()
   275→   275→    this._sortedNodes = []
   276→   276→    this._sortedDirty = true
   277→   277→
   278→   278→    if (!categories.length || !terms.length) return
   279→   279→
   280→   280→    // Pre-build category index and group terms by category (O(n) instead of O(n²))
   281→   281→    const catMap = new Map<string, GraphCategory>()
   282→   282→    for (let ci = 0; ci < categories.length; ci++) catMap.set(categories[ci].id, categories[ci])
   283→   283→    const catTermsMap = new Map<string, { term: GraphNode['term']; idx: number }[]>()
   284→   284→    for (let ti = 0; ti < terms.length; ti++) {
   285→   285→      const t = terms[ti]
   286→   286→      const arr = catTermsMap.get(t.category)
   287→   287→      if (arr) arr.push({ term: t, idx: ti })
   288→   288→      else catTermsMap.set(t.category, [{ term: t, idx: ti }])
   289→   289→    }
   290→   290→
   291→   291→    terms.forEach((term, termGlobalIdx) => {
   292→   292→      const category = catMap.get(term.category)
   293→   293→      const categoryIndex = category ? categories.indexOf(category) : 0
   294→   294→      const total = categories.length || 1
   295→   295→
   296→   296→      const baseAngle = (categoryIndex / total) * Math.PI * 2 - Math.PI / 2
   297→   297→      const catTermsArr = catTermsMap.get(term.category)
   298→   298→      const termIndex = catTermsArr ? catTermsArr.findIndex((e) => e.idx === termGlobalIdx) : 0
   299→   299→
   300→   300→      const radius = this.options.nodeRadius[term.type] ?? 16
   301→   301→      const distance = term.type === 'core' ? 160 : term.type === 'technique' ? 220 : 280
   302→   302→
   303→   303→      const spreadAngle = Math.PI / 4
   304→   304→      const catTermsLen = catTermsArr ? catTermsArr.length : 1
   305→   305→      const angleOffset =
   306→   306→        catTermsLen > 1
   307→   307→          ? ((termIndex - (catTermsLen - 1) / 2) * spreadAngle) / catTermsLen
   308→   308→          : 0
   309→   309→      const angle = baseAngle + angleOffset
   310→   310→
   311→   311→      const jx = (Math.random() - 0.5) * 40
   312→   312→      const jy = (Math.random() - 0.5) * 40
   313→   313→
   314→   314→      const node: GraphNode = {
   315→   315→        id: term.id,
   316→   316→        x: this.centerX + Math.cos(angle) * distance + jx,
   317→   317→        y: this.centerY + Math.sin(angle) * distance + jy,
   318→   318→        vx: 0,
   319→   319→        vy: 0,
   320→   320→        radius,
   321→   321→        term,
   322→   322→        color: category?.color ?? '#94a3b8',
   323→   323→        highlighted: false,
   324→   324→        visible: true,
   325→   325→        currentRadius: radius,
   326→   326→        targetRadius: radius,
   327→   327→      }
   328→   328→
   329→   329→      this.nodes.push(node)
   330→   330→      this.nodeMap.set(term.id, node)
   331→   331→    })
   332→   332→
   333→   333→    const edgeSet = new Set<string>()
   334→   334→    terms.forEach((term) => {
   335→   335→      const relatedIds = (term as unknown as { related?: string[] }).related ?? []
   336→   336→      relatedIds.forEach((relId: string) => {
   337→   337→        const key = [term.id, relId].sort().join('|')
   338→   338→        if (!edgeSet.has(key)) {
   339→   339→          edgeSet.add(key)
   340→   340→          this.edges.push({ source: term.id, target: relId, strength: 1 })
   341→   341→        }
   342→   342→      })
   343→   343→    })
   344→   344→
   345→   345→    this._warmUpPhysics(120)
   346→   346→  }
   347→   347→
   348→   348→  private _warmUpPhysics(iterations: number) {
   349→   349→    const CHUNK = 60
   350→   350→    let done = 0
   351→   351→    const run = () => {
   352→   352→      const end = Math.min(done + CHUNK, iterations)
   353→   353→      for (; done < end; done++) this._simulatePhysics(0.15)
   354→   354→      if (done < iterations) {
   355→   355→        const id = setTimeout(run, 0)
   356→   356→        this._warmUpTimers.push(id)
   357→   357→      }
   358→   358→    }
   359→   359→    run()
   360→   360→  }

   362→   362→  private _simulatePhysics(dt = 1): boolean {
   363→   363→    if (!this.physics.enabled) return false
   364→   364→
   365→   365→    const nodes = this.nodes
   366→   366→    const CELL = 200
   367→   367→    const eps = 0.001
   368→   368→
   369→   369→    const grid = new Map<string, GraphNode[]>()
   370→   370→    const cellOf = (x: number, y: number) => `${Math.floor(x / CELL)},${Math.floor(y / CELL)}`
   371→   371→
   372→   372→    nodes.forEach((node) => {
   373→   373→      if (!node.visible) return
   374→   374→      const key = cellOf(node.x, node.y)
   375→   375→      if (!grid.has(key)) grid.set(key, [])
   376→   376→      grid.get(key)!.push(node)
   377→   377→    })
   378→   378→
   379→   379→    nodes.forEach((node) => {
   380→   380→      if (!node.visible) return
   381→   381→
   382→   382→      node.vx += (this.centerX - node.x) * this.physics.centerGravity * dt
   383→   383→      node.vy += (this.centerY - node.y) * this.physics.centerGravity * dt
   384→   384→
   385→   385→      // Inline neighbor iteration — avoids allocating a new array per node per frame
   386→   386→      const cx = Math.floor(node.x / CELL)
   387→   387→      const cy = Math.floor(node.y / CELL)
   388→   388→      for (let dx = -1; dx <= 1; dx++) {
   389→   389→        for (let dy = -1; dy <= 1; dy++) {
   390→   390→          const key = `${cx + dx},${cy + dy}`
   391→   391→          const cell = grid.get(key)
   392→   392→          if (!cell) continue
   393→   393→          for (let ci = 0; ci < cell.length; ci++) {
   394→   394→            const other = cell[ci]
   395→   395→            if (node.id === other.id) continue
   396→   396→            const ddx = node.x - other.x
   397→   397→            const ddy = node.y - other.y
   398→   398→            const distSq = ddx * ddx + ddy * ddy
   399→   399→            const dist = Math.max(Math.sqrt(distSq), 1)
   400→   400→            if (dist < CELL) {
   401→   401→              const force = this.physics.repulsion / (distSq + eps)
   402→   402→              node.vx += (ddx / dist) * force * dt
   403→   403→              node.vy += (ddy / dist) * force * dt
   404→   404→            }
   405→   405→          }
   406→   406→        }
   407→   407→      }
   408→   408→    })
   409→   409→
   410→   410→    this.edges.forEach((edge) => {
   411→   411→      const src = this.nodeMap.get(edge.source)
   412→   412→      const tgt = this.nodeMap.get(edge.target)
   413→   413→      if (!src || !tgt || !src.visible || !tgt.visible) return
   414→   414→
   415→   415→      const dx = tgt.x - src.x
   416→   416→      const dy = tgt.y - src.y
   417→   417→      const dist = Math.max(Math.sqrt(dx * dx + dy * dy), 1)
   418→   418→      const targetDist = 180
   419→   419→      const force = (dist - targetDist) * this.physics.attraction
   420→   420→
   421→   421→      src.vx += (dx / dist) * force * dt
   422→   422→      src.vy += (dy / dist) * force * dt
   423→   423→      tgt.vx -= (dx / dist) * force * dt
   424→   424→      tgt.vy -= (dy / dist) * force * dt
   425→   425→    })
   426→   426→
   427→   427→    const maxV = this.physics.maxVelocity
   428→   428→    const padding = this.options.padding
   429→   429→    const minV = this.physics.minVelocity
   430→   430→    let moved = false
   431→   431→
   432→   432→    nodes.forEach((node) => {
   433→   433→      if (!node.visible) return
   434→   434→
   435→   435→      node.vx *= this.physics.damping
   436→   436→      node.vy *= this.physics.damping
   437→   437→
   438→   438→      const v = Math.sqrt(node.vx * node.vx + node.vy * node.vy)
   439→   439→      if (v > maxV) {
   440→   440→        node.vx = (node.vx / v) * maxV
   441→   441→        node.vy = (node.vy / v) * maxV
   442→   442→      }
   443→   443→
   444→   444→      if (Math.abs(node.vx) > minV) { node.x += node.vx; moved = true }
   445→   445→      if (Math.abs(node.vy) > minV) { node.y += node.vy; moved = true }
   446→   446→
   447→   447→      node.x = Math.max(padding, Math.min(this.width - padding, node.x))
   448→   448→      node.y = Math.max(padding, Math.min(this.height - padding, node.y))
   449→   449→
   450→   450→      const radiusDelta = (node.targetRadius - node.currentRadius) * 0.12
   451→   451→      if (Math.abs(radiusDelta) > 0.01) {
   452→   452→        node.currentRadius += radiusDelta
   453→   453→        moved = true
   454→   454→      }
   455→   455→    })
   456→   456→
   457→   457→    return moved
   458→   458→  }

   464→   464→  private findNodeAt(wx: number, wy: number, extra = 0): GraphNode | null {
   465→   465→    for (let i = this.nodes.length - 1; i >= 0; i--) {
   466→   466→      const node = this.nodes[i]
   467→   467→      if (!node.visible) continue
   468→   468→      const dx = wx - node.x
   469→   469→      const dy = wy - node.y
   470→   470→      if (Math.sqrt(dx * dx + dy * dy) <= node.radius + 5 + extra) return node
   471→   471→    }
   472→   472→    return null
   473→   473→  }

   488→   488→      this._panTargetX = this.panX
   489→   489→      this._panTargetY = this.panY
   490→   490→      this.lastMouse = { x: e.clientX, y: e.clientY }
   491→   491→    } else {
   492→   492→      const world = this.screenToWorld(x, y)
   493→   493→      const hovered = this.findNodeAt(world.x, world.y)
   494→   494→      if (hovered !== this.hoveredNode) {
   495→   495→        this.hoveredNode = hovered
   496→   496→        this.canvas!.style.cursor = hovered ? 'pointer' : 'grab'
   497→   497→        this._isDirty = true
   498→   498→        if (this.onHoverChange) this.onHoverChange(hovered, e)
   499→   499→      }
   500→   500→    }
   501→   501→  }
   502→   502→
   503→   503→  private _onMouseDown = (e: MouseEvent) => {
   504→   504→    this._wake()
   505→   505→    this.isDragging = true
   506→   506→    this._mouseDownPos = { x: e.clientX, y: e.clientY }
   507→   507→    this.lastMouse = { x: e.clientX, y: e.clientY }
   508→   508→    this.canvas!.style.cursor = 'grabbing'
   509→   509→  }
   510→   510→
   511→   511→  private _onMouseUp = () => {
   512→   512→    this.isDragging = false
   513→   513→    this.canvas!.style.cursor = this.hoveredNode ? 'pointer' : 'grab'
   514→   514→  }
   515→   515→
   516→   516→  private _onMouseLeave = () => {
   517→   517→    this.isDragging = false
   518→   518→    this.hoveredNode = null
   519→   519→    if (this.onHoverChange) this.onHoverChange(null, null)
   520→   520→  }
   521→   521→
   522→   522→  private _onWheel = (e: WheelEvent) => {
   523→   523→    this._wake()
   524→   524→    e.preventDefault()
   525→   525→    const { x, y } = this._clientPos(e)
   526→   526→    const delta = e.deltaY > 0 ? 0.9 : 1.1
   527→   527→    const newZoom = Math.max(this.options.zoomMin, Math.min(this.options.zoomMax, this.zoom * delta))
   528→   528→    this._animateZoom(newZoom, x, y)
   529→   529→  }
   530→   530→
   531→   531→  private _onClick = (e: MouseEvent) => {
   532→   532→    if (e.clientX === undefined && e.clientY === undefined) return
   533→   533→    // If mouse moved more than 5px from mousedown position, this was a drag/pan — skip click
   534→   534→    if (this._mouseDownPos) {
   535→   535→      const dx = e.clientX - this._mouseDownPos.x
   536→   536→      const dy = e.clientY - this._mouseDownPos.y
   537→   537→      if (Math.sqrt(dx * dx + dy * dy) > 5) return
   538→   538→    }
   539→   539→    const { x, y } = this._clientPos(e)
   540→   540→    const world = this.screenToWorld(x, y)
   541→   541→    const node = this.findNodeAt(world.x, world.y)
   542→   542→
   543→   543→    if (node) {
   544→   544→      this.selectedNode = node
   545→   545→      node.targetRadius = node.radius * 1.2
   546→   546→      setTimeout(() => {
   547→   547→        if (this.selectedNode === node) node.targetRadius = node.radius
   548→   548→      }, 220)
   549→   549→      if (this.onNodeSelect) this.onNodeSelect(node.term)
   550→   550→    } else {
   551→   551→      this.selectedNode = null
   552→   552→    }
   553→   553→  }
   554→   554→
   555→   555→  private _onTouchStart = (e: TouchEvent) => {
   556→   556→    this._wake()
   557→   557→    if (e.touches.length === 1) {
   558→   558→      e.preventDefault()
   559→   559→      const t = e.touches[0]
   560→   560→      const rect = this.canvas!.getBoundingClientRect()
   561→   561→      this._touchStartPos = { x: t.clientX - rect.left, y: t.clientY - rect.top }
   562→   562→      this._touchStartTime = performance.now()
   563→   563→      this.isDragging = true
   564→   564→      this.lastMouse = { x: t.clientX, y: t.clientY }
   565→   565→    } else if (e.touches.length === 2) {
   566→   566→      e.preventDefault()
   567→   567→      this.isDragging = false
   568→   568→      const dx = e.touches[0].clientX - e.touches[1].clientX
   569→   569→      const dy = e.touches[0].clientY - e.touches[1].clientY
   570→   570→      this._pinchDist = Math.sqrt(dx * dx + dy * dy)
   571→   571→      this._pinchZoom = this.zoom
   572→   572→    }
   573→   573→  }
   574→   574→
   575→   575→  private _onTouchMove = (e: TouchEvent) => {
   576→   576→    if (e.touches.length === 1 && this.isDragging) {
   577→   577→      e.preventDefault()
   578→   578→      const t = e.touches[0]
   579→   579→      const dx = t.clientX - this.lastMouse.x
   580→   580→      const dy = t.clientY - this.lastMouse.y
   581→   581→      this.panX += dx
   582→   582→      this.panY += dy
   583→   583→      this._panTargetX = this.panX
   584→   584→      this._panTargetY = this.panY
   585→   585→      this.lastMouse = { x: t.clientX, y: t.clientY }
   586→   586→    } else if (e.touches.length === 2 && this._pinchDist) {
   587→   587→      e.preventDefault()
   588→   588→      const dx = e.touches[0].clientX - e.touches[1].clientX
   589→   589→      const dy = e.touches[0].clientY - e.touches[1].clientY
   590→   590→      const dist = Math.sqrt(dx * dx + dy * dy)
   591→   591→      const factor = dist / this._pinchDist
   592→   592→      const newZoom = Math.max(this.options.zoomMin, Math.min(this.options.zoomMax, this._pinchZoom * factor))
   593→   593→      this._animateZoom(newZoom, this.width / 2, this.height / 2)
   594→   594→    }
   595→   595→  }
   596→   596→
   597→   597→  private _onTouchEnd = (e: TouchEvent) => {
   598→   598→    if (e.changedTouches.length > 0 && this._touchStartPos) {
   599→   599→      const t = e.changedTouches[0]
   600→   600→      const rect = this.canvas!.getBoundingClientRect()
   601→   601→      const ex = t.clientX - rect.left
   602→   602→      const ey = t.clientY - rect.top
   603→   603→
   604→   604→      const distMoved = Math.sqrt((ex - this._touchStartPos.x) ** 2 + (ey - this._touchStartPos.y) ** 2)
   605→   605→      const elapsed = performance.now() - this._touchStartTime
   606→   606→
   607→   607→      if (distMoved < this._TAP_MAX_DIST && elapsed < this._TAP_MAX_MS) {
   608→   608→        const world = this.screenToWorld(ex, ey)
   609→   609→        const node = this.findNodeAt(world.x, world.y, 12)
   610→   610→        if (node) {
   611→   611→          this.hoveredNode = node
   612→   612→          this.selectedNode = node
   613→   613→          node.targetRadius = node.radius * 1.2
   614→   614→          setTimeout(() => {
   615→   615→            if (this.selectedNode === node) node.targetRadius = node.radius
   616→   616→          }, 220)
   617→   617→          if (this.onNodeSelect) this.onNodeSelect(node.term)
   618→   618→          if (this.onHoverChange) this.onHoverChange(node, e)
   619→   619→        } else {
   620→   620→          this.selectedNode = null
   621→   621→          this.hoveredNode = null
   622→   622→          if (this.onHoverChange) this.onHoverChange(null, null)
   623→   623→        }
   624→   624→      }
   625→   625→    }
   626→   626→
   627→   627→    this.isDragging = false
   628→   628→    this._pinchDist = null
   629→   629→    this._touchStartPos = null
   630→   630→    this.canvas!.style.cursor = 'grab'
   631→   631→  }
   632→   632→
   633→   633→  private _bindEvents() {
   634→   634→    if (!this.canvas) return
   635→   635→    const add = (target: EventTarget, type: string, fn: EventListener, opts?: AddEventListenerOptions) => {
   636→   636→      target.addEventListener(type, fn, opts)
   637→   637→      this._boundListeners.push({ target, type, fn, opts })
   638→   638→    }
   639→   639→
   640→   640→    add(window, 'resize', () => {
   641→   641→      if (this._resizeTimer) clearTimeout(this._resizeTimer)
   642→   642→      this._resizeTimer = setTimeout(() => this._handleResize(), 200)
   643→   643→    })
   644→   644→    add(this.canvas, 'mousemove', this._onMouseMove as EventListener)
   645→   645→    add(this.canvas, 'mousedown', this._onMouseDown as EventListener)
   646→   646→    add(this.canvas, 'mouseup', this._onMouseUp as EventListener)
   647→   647→    add(this.canvas, 'mouseleave', this._onMouseLeave as EventListener)
   648→   648→    add(this.canvas, 'wheel', this._onWheel as EventListener, { passive: false })
   649→   649→    add(this.canvas, 'click', this._onClick as EventListener)
   650→   650→    add(this.canvas, 'touchstart', this._onTouchStart as EventListener, { passive: false })
   651→   651→    add(this.canvas, 'touchmove', this._onTouchMove as EventListener, { passive: false })
   652→   652→    add(this.canvas, 'touchend', this._onTouchEnd as EventListener)
   653→   653→
   654→   654→    add(document, 'visibilitychange', () => {
   655→   655→      if (document.hidden) {
   656→   656→        this.stopAnimation()
   657→   657→      } else {
   658→   658→        this._lastTs = null
   659→   659→        this._startAnimation()
   660→   660→      }
   661→   661→    })
   662→   662→  }
   663→   663→
   664→   664→  private _handleResize() {
   665→   665→    if (!this.canvas) return
   666→   666→    const parent = this.canvas.parentElement
   667→   667→    if (!parent) return
   668→   668→    const rect = parent.getBoundingClientRect()
   669→   669→
   670→   670→    this.width = rect.width || 800
   671→   671→    this.height = rect.height || 600
   672→   672→    this.centerX = this.width / 2
   673→   673→    this.centerY = this.height / 2
   674→   674→    this.dpr = window.devicePixelRatio || 1
   675→   675→
   676→   676→    this.canvas.width = Math.round(this.width * this.dpr)
   677→   677→    this.canvas.height = Math.round(this.height * this.dpr)
   678→   678→    this.canvas.style.width = `${this.width}px`
   679→   679→    this.canvas.style.height = `${this.height}px`
   680→   680→
   681→   681→    this._gridDirty = true
   682→   682→    this._isDirty = true
   683→   683→  }
   684→   684→
   685→   685→  private _animateZoom(newZoom: number, pivotX: number, pivotY: number) {
   686→   686→    const scale = newZoom / this.zoom
   687→   687→    this._panTargetX = pivotX - (pivotX - this.panX) * scale
   688→   688→    this._panTargetY = pivotY - (pivotY - this.panY) * scale
   689→   689→    this._zoomTarget = newZoom
   690→   690→  }
   691→   691→
   692→   692→  private _startAnimation() {
   693→   693→    if (this._isRunning) return
   694→   694→
   695→   695→    const tick = (ts: number) => {
   696→   696→      const dt = this._lastTs ? Math.min((ts - this._lastTs) / 1000, 0.05) : 0.016
   697→   697→      this._lastTs = ts
   698→   698→
   699→   699→      this.time += dt
   700→   700→
   701→   701→      const LERP = 0.18
   702→   702→      const zooming = Math.abs(this._zoomTarget - this.zoom) > 0.0005
   703→   703→      const panning = Math.abs(this._panTargetX - this.panX) > 0.1 || Math.abs(this._panTargetY - this.panY) > 0.1
   704→   704→      if (zooming || panning) {
   705→   705→        this.zoom += (this._zoomTarget - this.zoom) * LERP
   706→   706→        this.panX += (this._panTargetX - this.panX) * LERP
   707→   707→        this.panY += (this._panTargetY - this.panY) * LERP
   708→   708→        this._isDirty = true
   709→   709→      }
   710→   710→
   711→   711→      const hadMovement = this._simulatePhysics(dt * 60)
   712→   712→      if (hadMovement) this._isDirty = true
   713→   713→
   714→   714→      // Skip render when idle (nothing changed for several frames)
   715→   715→      const shouldRender = this._isDirty || this.hoveredNode || this.selectedNode || this._idleFrames < 3
   716→   716→      if (shouldRender) {
   717→   717→        this._render()
   718→   718→        this._idleFrames = 0
   719→   719→      } else {
   720→   720→        this._idleFrames++
   721→   721→      }
   722→   722→
   723→   723→      this._isDirty = false
   724→   724→
   725→   725→      // Full stop when truly idle — saves CPU/battery
   726→   726→      if (this._idleFrames >= this._MAX_IDLE) {
   727→   727→        this._isRunning = false
   728→   728→        this.animationId = null
   729→   729→        return
   730→   730→      }
   731→   731→
   732→   732→      this.animationId = requestAnimationFrame(tick)
   733→   733→    }
   734→   734→
   735→   735→    this._isRunning = true
   736→   736→    this.animationId = requestAnimationFrame(tick)
   737→   737→  }
   738→   738→
   739→   739→  private _rebuildGridCache() {
   740→   740→    const w = this.width
   741→   741→    const h = this.height
   742→   742→    const cw = Math.round(w * this.dpr)
   743→   743→    const ch = Math.round(h * this.dpr)
   744→   744→
   745→   745→    // Reuse existing canvas if dimensions haven't changed
   746→   746→    if (!this._gridCanvas) {
   747→   747→      this._gridCanvas = document.createElement('canvas')
   748→   748→      this._gridCanvas.width = cw
   749→   749→      this._gridCanvas.height = ch
   750→   750→    } else if (this._gridCanvas.width !== cw || this._gridCanvas.height !== ch) {
   751→   751→      this._gridCanvas.width = cw
   752→   752→      this._gridCanvas.height = ch
   753→   753→    }
   754→   754→    // else: dimensions match, just repaint
   755→   755→
   756→   756→    const gc = this._gridCanvas.getContext('2d')!
   757→   757→    gc.setTransform(this.dpr, 0, 0, this.dpr, 0, 0)
   758→   758→    gc.fillStyle = this.theme.gridDot
   759→   759→
   760→   760→    const gap = 40
   761→   761→    for (let x = gap; x < w; x += gap) {
   762→   762→      for (let y = gap; y < h; y += gap) {
   763→   763→        gc.beginPath()
   764→   764→        gc.arc(x, y, 1.5, 0, Math.PI * 2)
   765→   765→        gc.fill()
   766→   766→      }
   767→   767→    }
   768→   768→
   769→   769→    this._gridDirty = false
   770→   770→  }
   771→   771→
   772→   772→  private _render() {
   773→   773→    const ctx = this.ctx
   774→   774→    if (!ctx || !this.canvas) return
   775→   775→
   776→   776→    // Track selection changes for sort optimization
   777→   777→    const curSelId = this.selectedNode?.id ?? null
   778→   778→    if (curSelId !== this._prevSelectedId) {
   779→   779→      this._sortedDirty = true
   780→   780→      this._prevSelectedId = curSelId
   781→   781→    }
   782→   782→
   783→   783→    ctx.setTransform(this.dpr, 0, 0, this.dpr, 0, 0)
   784→   784→
   785→   785→    const bgGrad = ctx.createRadialGradient(this.centerX, this.centerY, 0, this.centerX, this.centerY, Math.max(this.width, this.height) * 0.7)
   786→   786→    bgGrad.addColorStop(0, this.theme.bgGradientCenter)
   787→   787→    bgGrad.addColorStop(1, this.theme.bg)
   788→   788→    ctx.fillStyle = bgGrad
   789→   789→    ctx.fillRect(0, 0, this.width, this.height)
   790→   790→
   791→   791→    if (this._gridDirty || !this._gridCanvas) this._rebuildGridCache()
   792→   792→    if (this._gridCanvas) ctx.drawImage(this._gridCanvas, 0, 0, this.width, this.height)
   793→   793→
   794→   794→    ctx.save()
   795→   795→    ctx.translate(this.panX, this.panY)
   796→   796→    ctx.scale(this.zoom, this.zoom)
   797→   797→
   798→   798→    this._drawEdges(ctx)
   799→   799→    this._drawNodes(ctx)
   800→   800→
   801→   801→    ctx.restore()
   802→   802→  }
   803→   803→
   804→   804→  private _drawEdges(ctx: CanvasRenderingContext2D) {
   805→   805→    const selected = this.selectedNode
   806→   806→    const hovered = this.hoveredNode
   807→   807→
   808→   808→    this.edges.forEach((edge) => {
   809→   809→      const src = this.nodeMap.get(edge.source)
   810→   810→      const tgt = this.nodeMap.get(edge.target)
   811→   811→      if (!src || !tgt || !src.visible || !tgt.visible) return
   812→   812→
   813→   813→      const srcActive = selected && (selected.id === src.id || selected.id === tgt.id)
   814→   814→      const hovActive = hovered && (hovered.id === src.id || hovered.id === tgt.id)
   815→   815→
   816→   816→      let opacity: number
   817→   817→      if (selected) {
   818→   818→        opacity = srcActive ? 1.0 : 0.06
   819→   819→      } else if (hovered) {
   820→   820→        opacity = hovActive ? 0.85 : 0.12
   821→   821→      } else {
   822→   822→        opacity = 0.55
   823→   823→      }
   824→   824→
   825→   825→      const dx = tgt.x - src.x
   826→   826→      const dy = tgt.y - src.y
   827→   827→      const dist = Math.max(Math.sqrt(dx * dx + dy * dy), 1)
   828→   828→
   829→   829→      const mx = (src.x + tgt.x) / 2
   830→   830→      const my = (src.y + tgt.y) / 2
   831→   831→      // Edge curve direction: since edges are bidirectional (no inherent direction),
   832→   832→      // we use a deterministic but arbitrary ID-based comparison so that the curve bends
   833→   833→      // consistently for each pair rather than randomly flipping.
   834→   834→      const off = dist * 0.15
   835→   835→      const dir = src.id < tgt.id ? 1 : -1
   836→   836→      const cx = mx + (dy / dist) * off * dir
   837→   837→      const cy = my - (dx / dist) * off * dir
   838→   838→
   839→   839→      const grad = ctx.createLinearGradient(src.x, src.y, tgt.x, tgt.y)
   840→   840→      grad.addColorStop(0, this._hexToRgba(src.color, opacity))
   841→   841→      grad.addColorStop(1, this._hexToRgba(tgt.color, opacity))
   842→   842→
   843→   843→      ctx.beginPath()
   844→   844→      ctx.moveTo(src.x, src.y)
   845→   845→      ctx.quadraticCurveTo(cx, cy, tgt.x, tgt.y)
   846→   846→      ctx.strokeStyle = grad
   847→   847→      ctx.lineWidth = srcActive || hovActive ? 2.5 : 1.5
   848→   848→      ctx.lineCap = 'round'
   849→   849→      ctx.setLineDash([])
   850→   850→      ctx.stroke()
   851→   851→
   852→   852→      // Arrowhead
   853→   853→      const tangX = tgt.x - cx
   854→   854→      const tangY = tgt.y - cy
   855→   855→      const angle = Math.atan2(tangY, tangX)
   856→   856→      const arrowX = tgt.x - Math.cos(angle) * (tgt.radius + 6)
   857→   857→      const arrowY = tgt.y - Math.sin(angle) * (tgt.radius + 6)
   858→   858→
   859→   859→      ctx.save()
   860→   860→      ctx.translate(arrowX, arrowY)
   861→   861→      ctx.rotate(angle)
   862→   862→      ctx.beginPath()
   863→   863→      ctx.moveTo(0, 0)
   864→   864→      ctx.lineTo(-8, 4)
   865→   865→      ctx.lineTo(-8, -4)
   866→   866→      ctx.closePath()
   867→   867→      ctx.fillStyle = this._hexToRgba(tgt.color, opacity)
   868→   868→      ctx.fill()
   869→   869→      ctx.restore()
   870→   870→
   871→   871→      // Quantum packets on active edges
   872→   872→      if (srcActive || hovActive) {
   873→   873→        ctx.save()
   874→   874→        ctx.shadowBlur = 6
   875→   875→        ctx.shadowColor = this._isDark ? '#06b6d4' : '#06b6d4'
   876→   876→
   877→   877→        const packetColor = this._isDark ? '#22d3ee' : '#06b6d4'
   878→   878→        for (let i = 0; i < 3; i++) {
   879→   879→          const t = ((this.time * 0.6 + i / 3)) % 1
   880→   880→          const pos = this._bezierPoint(t, src.x, src.y, cx, cy, tgt.x, tgt.y)
   881→   881→          const sz = 2 + Math.sin(this.time * 10 + i) * 0.5
   882→   882→
   883→   883→          ctx.beginPath()
   884→   884→          ctx.arc(pos.x, pos.y, sz, 0, Math.PI * 2)
   885→   885→          ctx.fillStyle = packetColor
   886→   886→          ctx.fill()
   887→   887→        }
   888→   888→
   889→   889→        ctx.restore()
   890→   890→        ctx.shadowBlur = 0
   891→   891→      }
   892→   892→    })
   893→   893→  }
   894→   894→
   895→   895→  private _drawNodes(ctx: CanvasRenderingContext2D) {
   896→   896→    // Re-sort when selection changes — separate sorting from drawing to avoid mutation
   897→   897→    if (this._sortedDirty) {
   898→   898→      this._sortedNodes = this.nodes.slice()
   899→   899→      // Move selected node to end for draw order (on top)
   900→   900→      const selId = this.selectedNode?.id
   901→   901→      if (selId) {
   902→   902→        const selIdx = this._sortedNodes.findIndex((n) => n.id === selId)
   903→   903→        if (selIdx !== -1 && selIdx !== this._sortedNodes.length - 1) {
   904→   904→          const [selNode] = this._sortedNodes.splice(selIdx, 1)
   905→   905→          this._sortedNodes.push(selNode)
   906→   906→        }
   907→   907→      }
   908→   908→      this._sortedDirty = false
   909→   909→    }
   910→   910→
   911→   911→    const nodes = this._sortedNodes
   912→   912→    const selId = this.selectedNode?.id
   913→   913→    const hovId = this.hoveredNode?.id
   914→   914→    const hasSelection = !!selId
   915→   915→    const fontSize = this.options.fontSize
   916→   916→    const fontFamily = "'JetBrains Mono', ui-monospace, monospace"
   917→   917→    const time = this.time
   918→   918→    const isDark = this._isDark
   919→   919→
   920→   920→    for (let ni = 0; ni < nodes.length; ni++) {
   921→   921→      const node = nodes[ni]
   922→   922→      if (!node.visible) continue
   923→   923→
   924→   924→      const isSel = selId === node.id
   925→   925→      const isHov = hovId === node.id
   926→   926→      const dimmed = hasSelection && !isSel && !node.highlighted
   927→   927→
   928→   928→      let r = Math.max(1, node.currentRadius + (isSel ? Math.sin(time * 4) * 1.5 : 0))
   929→   929→
   930→   930→      // Outer scanning rings (selected)
   931→   931→      if (isSel) {
   932→   932→        ctx.save()
   933→   933→        ctx.translate(node.x, node.y)
   934→   934→        ctx.globalAlpha = 0.6
   935→   935→        ctx.strokeStyle = node.color
   936→   936→        ctx.lineWidth = 1.5
   937→   937→        ctx.setLineDash([4, 8])
   938→   938→        ctx.rotate(-time)
   939→   939→        ctx.beginPath()
   940→   940→        ctx.arc(0, 0, r + 8, 0, Math.PI * 2)
   941→   941→        ctx.stroke()
   942→   942→
   943→   943→        ctx.setLineDash([])
   944→   944→        ctx.lineWidth = 1
   945→   945→        ctx.globalAlpha = 0.3 + Math.sin(time * 5) * 0.2
   946→   946→        ctx.rotate(time * 2)
   947→   947→        ctx.beginPath()
   948→   948→        ctx.arc(0, 0, r + 4 + Math.sin(time * 3) * 3, 0, Math.PI * 2)
   949→   949→        ctx.stroke()
   950→   950→        ctx.restore()
   951→   951→      }
   952→   952→
   953→   953→      // Highlighted ring (search result)
   954→   954→      if (node.highlighted && !isSel) {
   955→   955→        ctx.save()
   956→   956→        ctx.globalAlpha = 0.7 + Math.sin(time * 6) * 0.2
   957→   957→        ctx.strokeStyle = node.color
   958→   958→        ctx.lineWidth = 2
   959→   959→        ctx.setLineDash([3, 5])
   960→   960→        ctx.beginPath()
   961→   961→        ctx.arc(node.x, node.y, r + 5, 0, Math.PI * 2)
   962→   962→        ctx.stroke()
   963→   963→        ctx.setLineDash([])
   964→   964→        ctx.restore()
   965→   965→      }
   966→   966→
   967→   967→      // Shadow / glow
   968→   968→      ctx.shadowColor = isSel || isHov ? node.color : 'rgba(0,0,0,0.06)'
   969→   969→      ctx.shadowBlur = isSel || isHov ? 12 : 3
   970→   970→
   971→   971→      // Node body
   972→   972→      ctx.beginPath()
   973→   973→      ctx.arc(node.x, node.y, r, 0, Math.PI * 2)
   974→   974→
   975→   975→      const innerR = Math.max(r * 0.1, 0.5)
   976→   976→      const grad = ctx.createRadialGradient(node.x - r * 0.3, node.y - r * 0.3, innerR, node.x, node.y, Math.max(r, 1))
   977→   977→
   978→   978→      if (isSel || isHov || node.highlighted) {
   979→   979→        grad.addColorStop(0, isDark ? '#334155' : '#ffffff')
   980→   980→        grad.addColorStop(0.4, this._hexToRgba(node.color, dimmed ? 0.3 : 1))
   981→   981→        grad.addColorStop(1, this._darkenColor(node.color, 20))
   982→   982→      } else {
   983→   983→        grad.addColorStop(0, dimmed ? this.theme.dimmedBodyStart : this.theme.nodeBodyStart)
   984→   984→        grad.addColorStop(1, dimmed ? this.theme.dimmedBodyEnd : this.theme.nodeBodyEnd)
   985→   985→      }
   986→   986→
   987→   987→      ctx.globalAlpha = dimmed ? 0.35 : 1
   988→   988→      ctx.fillStyle = grad
   989→   989→      ctx.fill()
   990→   990→      ctx.globalAlpha = 1
   991→   991→      ctx.shadowBlur = 0
   992→   992→
   993→   993→      // Border
   994→   994→      ctx.strokeStyle = isSel || isHov ? node.color : this.theme.nodeBorderDefault
   995→   995→      ctx.lineWidth = isSel || isHov ? 2.5 : 1
   996→   996→      ctx.stroke()
   997→   997→
   998→   998→      // Label (with cache)
   999→   999→      const maxLabelPx = Math.round((r - 3) * 2)
  1000→  1000→      const cacheKey = `${node.id}:${maxLabelPx}`
  1001→  1001→      let label = this._labelCache.get(cacheKey)
  1002→  1002→      if (label === undefined) {
  1003→  1003→        ctx.font = `600 ${fontSize}px ${fontFamily}`
  1004→  1004→        label = this._truncateLabel(ctx, node.term.name, maxLabelPx)
  1005→  1005→        this._labelCache.set(cacheKey, label)
  1006→  1006→      }
  1007→  1007→      ctx.font = `600 ${fontSize}px ${fontFamily}`
  1008→  1008→      ctx.textAlign = 'center'
  1009→  1009→      ctx.textBaseline = 'middle'
  1010→  1010→      ctx.fillStyle = isSel ? (isDark ? '#f4f4f5' : '#111827') : isHov ? node.color : this.theme.nodeLabelDefault
  1011→  1011→
  1012→  1012→      ctx.shadowColor = 'rgba(255,255,255,0.85)'
  1013→  1013→      ctx.shadowBlur = 4
  1014→  1014→      ctx.fillText(label, node.x, node.y)
  1015→  1015→      ctx.shadowBlur = 0
  1016→  1016→    }
  1017→  1017→  }

  1021→  1021→  private _truncateLabel(ctx: CanvasRenderingContext2D, text: string, maxPx: number): string {
  1022→  1022→    if (ctx.measureText(text).width <= maxPx) return text
  1023→  1023→    let t = text
  1024→  1024→    while (t.length > 1 && ctx.measureText(t + '\u2026').width > maxPx) {
  1025→  1025→      t = t.slice(0, -1)
  1026→  1026→    }
  1027→  1027→    return t + '\u2026'
  1028→  1028→  }

  1030→  1030→  private _bezierPoint(t: number, sx: number, sy: number, cx: number, cy: number, tx: number, ty: number) {
  1031→  1031→    const u = 1 - t
  1032→  1032→    return {
  1033→  1033→      x: u * u * sx + 2 * u * t * cx + t * t * tx,
  1034→  1034→      y: u * u * sy + 2 * u * t * cy + t * t * ty,
  1035→  1035→    }

  1038→  1038→  private _hexToRgba(hex: string, alpha = 1): string {
  1039→  1039→    if (typeof hex !== 'string' || !hex.startsWith('#')) return `rgba(148,163,184,${alpha})`
  1040→  1040→
  1041→  1041→    let h = hex.slice(1)
  1042→  1042→    if (h.length === 3) h = h[0] + h[0] + h[1] + h[1] + h[2] + h[2]
  1043→  1043→    if (h.length !== 6) return `rgba(148,163,184,${alpha})`
  1044→  1044→
  1045→  1045→    const r = parseInt(h.slice(0, 2), 16)
  1046→  1046→    const g = parseInt(h.slice(2, 4), 16)
  1047→  1047→    const b = parseInt(h.slice(4, 6), 16)
  1048→  1048→
  1049→  1049→    if (isNaN(r) || isNaN(g) || isNaN(b)) return `rgba(148,163,184,${alpha})`
  1050→  1050→    return `rgba(${r},${g},${b},${alpha})`
  1051→  1051→  }

  1053→  1053→  private _darkenColor(hex: string, percent: number): string {
  1054→  1054→    if (typeof hex !== 'string' || !hex.startsWith('#')) return '#94a3b8'
  1055→  1055→
  1056→  1056→    let h = hex.slice(1)
  1057→  1057→    if (h.length === 3) h = h[0] + h[0] + h[1] + h[1] + h[2] + h[2]
  1058→  1058→    if (h.length !== 6) return '#94a3b8'
  1059→  1059→
  1060→  1060→    const amt = Math.round(2.55 * percent)
  1061→  1061→    const num = parseInt(h, 16)
  1062→  1062→    const R = Math.max(0, (num >> 16) - amt)
  1063→  1063→    const G = Math.max(0, ((num >> 8) & 0x00ff) - amt)
  1064→  1064→    const B = Math.max(0, (num & 0xff) - amt)
  1065→  1065→    return '#' + (0x1000000 + R * 0x10000 + G * 0x100 + B).toString(16).slice(1)
  1066→  1066→  }
