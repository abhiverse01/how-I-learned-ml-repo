     8→     8→function escapeHTML(text: string): string {
     9→     9→  return String(text ?? '')
    10→    10→    .replace(/&/g, '&amp;')
    11→    11→    .replace(/</g, '&lt;')
    12→    12→    .replace(/>/g, '&gt;')
    13→    13→    .replace(/"/g, '&quot;')
    14→    14→    .replace(/'/g, '&#39;')
    15→    15→}

    17→    17→function parseMarkdown(text: string): string {
    18→    18→  if (!text) return ''
    19→    19→  let out = escapeHTML(text)
    20→    20→  out = out.replace(/```([\s\S]*?)```/g, (_, code) => `<pre><code>${code.trim()}</code></pre>`)
    21→    21→  out = out.replace(/`([^`\n]+)`/g, '<code>$1</code>')
    22→    22→  out = out.replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>')
    23→    23→  out = out.replace(/(?<!\*)\*(?!\*)(.+?)(?<!\*)\*(?!\*)/g, '<em>$1</em>')
    24→    24→  out = out.replace(/^### (.+)$/gm, '<h4>$1</h4>')
    25→    25→  out = out.replace(/^## (.+)$/gm, '<h3>$1</h3>')
    26→    26→  out = out.replace(/^# (.+)$/gm, '<h2>$1</h2>')
    27→    27→  out = out.replace(
    28→    28→    /(?:^- .+\n?)+/gm,
    29→    29→    (block) =>
    30→    30→      '<ul>' +
    31→    31→      block
    32→    32→        .trim()
    33→    33→        .split('\n')
    34→    34→        .map((l) => `<li>${l.replace(/^- /, '').trim()}</li>`)
    35→    35→        .join('') +
    36→    36→      '</ul>'
    37→    37→  )
    38→    38→  out = out.replace(
    39→    39→    /(?:^\d+\. .+\n?)+/gm,
    40→    40→    (block) =>
    41→    41→      '<ol>' +
    42→    42→      block
    43→    43→        .trim()
    44→    44→        .split('\n')
    45→    45→        .map((l) => `<li>${l.replace(/^\d+\. /, '').trim()}</li>`)
    46→    46→        .join('') +
    47→    47→      '</ol>'
    48→    48→  )
    49→    49→  const BLOCK_START = /^<(ul|ol|pre|h[1-6]|blockquote)/
    50→    50→  out = out
    51→    51→    .split(/\n{2,}/)
    52→    52→    .map((block) => {
    53→    53→      const trimmed = block.trim()
    54→    54→      if (!trimmed) return ''
    55→    55→      if (BLOCK_START.test(trimmed)) return trimmed
    56→    56→      return `<p>${trimmed.replace(/\n/g, '<br>')}</p>`
    57→    57→    })
    58→    58→    .filter(Boolean)
    59→    59→    .join('')
    60→    60→  return out
    61→    61→}

    63→    63→// ─── Architecture step renderer ───
    64→    64→function renderArchSteps(steps: { label: string; desc?: string; type?: string; inputId?: string; children?: typeof steps }[]): string {
    65→    65→  if (!steps || !steps.length) return ''
    66→    66→  return steps
    67→    67→    .map((step) => {
    68→    68→      let typeClass = `step-type-${step.type || 'process'}`
    69→    69→      let childrenHtml = ''
    70→    70→      if (step.children && step.children.length > 0) {
    71→    71→        typeClass += ' step-type-container'
    72→    72→        childrenHtml = `<div class="step-children">${renderArchSteps(step.children)}</div>`
    73→    73→      }
    74→    74→      return `
    75→    75→        <div class="flow-step">
    76→    76→          <div class="step-box ${typeClass}">
    77→    77→            <div class="step-label">${escapeHTML(step.label)}</div>
    78→    78→            ${step.desc ? `<div class="step-desc">${escapeHTML(step.desc)}</div>` : ''}
    79→    79→          </div>
    80→    80→          ${childrenHtml}
    81→    81→        </div>
    82→    82→      `
    83→    83→    })
    84→    84→    .join('')
    85→    85→}

   117→   117→  const bg =
   118→   118→    toast.type === 'success'
   119→   119→      ? 'var(--accent-success, #10b981)'

   123→   123→
   124→   124→  return (
   125→   125→    <div
   126→   126→      key={toast.key}
   127→   127→      className={`app-toast ${toast.exiting ? 'toast-exit' : ''}`}
   128→   128→      style={{ background: bg }}
   129→   129→      role="status"
   130→   130→      aria-live="polite"
   131→   131→    >
   132→   132→      {toast.message}
   133→   133→    </div>

   441→   441→    setCopyState('idle')
   442→   442→  }, [selectedTerm])
   443→   443→
   444→   444→  // ─── Handlers ───
   445→   445→  const handleCategoryClick = useCallback((categoryId: string) => {
   446→   446→    if (selectedCategory === categoryId) {
   447→   447→      setSelectedCategory(null)
   448→   448→      graphRef.current?.filterByCategory(null)
   449→   449→    } else {
   450→   450→      setSelectedCategory(categoryId)
   451→   451→      graphRef.current?.filterByCategory(categoryId)
   452→   452→    }
   453→   453→  }, [selectedCategory, setSelectedCategory])
   454→   454→
   455→   455→  const handleFilterClick = useCallback((filter: string) => {
   456→   456→    setSelectedFilter(filter)
   457→   457→    if (graphRef.current) {
   458→   458→      for (let i = 0; i < graphRef.current.nodes.length; i++) {
   459→   459→        graphRef.current.nodes[i].visible = filter === 'all' || graphRef.current.nodes[i].term.type === filter
   460→   460→      }
   461→   461→      if (filter === 'all') setSelectedCategory(null)
   462→   462→    }
   463→   463→  }, [setSelectedFilter, setSelectedCategory])
   464→   464→
   465→   465→  const handleRelatedClick = useCallback((termId: string) => {
   466→   466→    const term = terms.find((t) => t.id === termId)
   467→   467→    if (term) {
   468→   468→      setSelectedTerm(term)
   469→   469→      navigateTerm(term.id)
   470→   470→    }
   471→   471→  }, [terms, setSelectedTerm, navigateTerm])
   472→   472→
   473→   473→  const handleAddTerm = useCallback((e: React.FormEvent<HTMLFormElement>) => {
   474→   474→    e.preventDefault()
   475→   475→    const formData = new FormData(e.currentTarget)
   476→   476→    const name = formData.get('name')?.toString().trim() ?? ''
   477→   477→    const categoryId = formData.get('category')?.toString().trim() ?? ''
   478→   478→    const shortDesc = formData.get('shortDesc')?.toString().trim() ?? ''
   479→   479→    const definition = formData.get('definition')?.toString().trim() ?? ''
   480→   480→    const relatedStr = formData.get('related')?.toString().trim() ?? ''
   481→   481→    const tagsStr = formData.get('tags')?.toString().trim() ?? ''
   482→   482→
   483→   483→    if (!name || !categoryId || !shortDesc || !definition) {
   484→   484→      showToast('Please fill all required fields.', 'error')
   485→   485→      return
   486→   486→    }
   487→   487→
   488→   488→    const baseId = name.toLowerCase().replace(/\s+/g, '-').replace(/[^a-z0-9-]/g, '')
   489→   489→    if (!baseId) {
   490→   490→      showToast('Could not generate a valid ID from that name.', 'error')
   491→   491→      return
   492→   492→    }
   493→   493→
   494→   494→    let id = baseId
   495→   495→    let suffix = 1
   496→   496→    while (terms.find((t) => t.id === id)) {
   497→   497→      id = `${baseId}-${++suffix}`
   498→   498→    }
   499→   499→
   500→   500→    const related = relatedStr
   501→   501→      .split(',')
   502→   502→      .map((s) => s.trim().toLowerCase().replace(/\s+/g, '-').replace(/[^a-z0-9-]/g, ''))
   503→   503→      .filter(Boolean)
   504→   504→
   505→   505→    const tags = tagsStr
   506→   506→      .split(',')
   507→   507→      .map((s) => s.trim())
   508→   508→      .filter(Boolean)
   509→   509→
   510→   510→    addTerm({
   511→   511→      id, name, fullName: name, category: categoryId, type: 'technique',
   512→   512→      shortDesc, definition, related, tags, codeExample: '',
   513→   513→      createdAt: new Date().toISOString(), importance: 0,
   514→   514→    })
   515→   515→
   516→   516→    if (graphRef.current) {
   517→   517→      const cats = categories.map((c) => ({ id: c.id, name: c.name, color: c.color }))
   518→   518→      const termsData = terms.map((t) => ({
   519→   519→        id: t.id, name: t.name, fullName: t.fullName,
   520→   520→        category: t.category, type: t.type, shortDesc: t.shortDesc,
   521→   521→        tags: t.tags, related: t.related,
   522→   522→      }))
   523→   523→      graphRef.current.loadData(cats, termsData)
   524→   524→    }
   525→   525→
   526→   526→    setAddModalOpen(false)
   527→   527→    showToast(`"${name}" added successfully!`, 'success')
   528→   528→  }, [addTerm, categories, terms, setAddModalOpen, showToast])
   529→   529→
   530→   530→  // Architecture handlers
   531→   531→  const showArchGallery = useCallback(() => {
   532→   532→    setActiveView('architecture')
   533→   533→    setArchCurrentView('gallery')
   534→   534→    setArchCurrentId(null)
   535→   535→  }, [setActiveView, setArchCurrentView, setArchCurrentId])
   536→   536→
   537→   537→  const showArchDetail = useCallback((archId: string) => {
   538→   538→    setArchCurrentId(archId)
   539→   539→    setArchCurrentView('architecture')
   540→   540→  }, [setArchCurrentId, setArchCurrentView])
   541→   541→
   542→   542→  const hideArchView = useCallback(() => {
   543→   543→    setActiveView('graph')
   544→   544→    setArchCurrentId(null)
   545→   545→  }, [setActiveView, setArchCurrentId])
   546→   546→
   547→   547→  // Knowledge path handlers
   548→   548→  const showPathsView = useCallback(() => {
   549→   549→    setActiveView('knowledge-paths')
   550→   550→    setPathMode('gallery')
   551→   551→    setActivePath(null)
   552→   552→    setPathSubPaths([])
   553→   553→  }, [setActiveView, setPathMode, setActivePath, setPathSubPaths])
   554→   554→
   555→   555→  const startPath = useCallback((path: KnowledgePath) => {
   556→   556→    if (path.paths && !path.steps) {
   557→   557→      setPathMode('subpaths')
   558→   558→      setPathSubPaths(path.paths)
   559→   559→      setActivePath(path)
   560→   560→    } else {
   561→   561→      setPathMode('steps')
   562→   562→      setActivePath(path)
   563→   563→    }
   564→   564→  }, [setPathMode, setPathSubPaths, setActivePath])
   565→   565→
   566→   566→  const focusTerm = useCallback((termName: string) => {
   567→   567→    setActiveView('graph')
   568→   568→    const term = terms.find((t) => t.name === termName || t.fullName === termName)
   569→   569→    if (term) {
   570→   570→      setSelectedTerm(term)
   571→   571→      navigateTerm(term.id)
   572→   572→    }
   573→   573→  }, [terms, setActiveView, setSelectedTerm, navigateTerm])
   574→   574→
   575→   575→  // ─── Memoized derived data ───
   576→   576→  const selectedTermCategory = useMemo(() => {
   577→   577→    if (!selectedTerm) return null
   578→   578→    return categories.find((c) => c.id === selectedTerm.category) ?? null
   579→   579→  }, [selectedTerm, categories])
   580→   580→
   581→   581→  const relatedTerms = useMemo(() => {
   582→   582→    if (!selectedTerm) return []
   583→   583→    const results: { term: Term; cat: Category | undefined }[] = []
   584→   584→    for (let i = 0; i < selectedTerm.related.length; i++) {
   585→   585→      const rt = terms.find((t) => t.id === selectedTerm.related[i])
   586→   586→      if (!rt) continue
   587→   587→      const rc = categories.find((c) => c.id === rt.category)
   588→   588→      results.push({ term: rt, cat: rc })
   589→   589→    }
   590→   590→    return results
   591→   591→  }, [selectedTerm, terms, categories])
   592→   592→
   593→   593→  const currentArch = useMemo(() => {
   594→   594→    if (!archCurrentId) return null
   595→   595→    return architectures.find((a) => a.id === archCurrentId) ?? null
   596→   596→  }, [archCurrentId, architectures])
   597→   597→
   598→   598→  // ─── Loading screen ───
   599→   599→  if (!isLoaded) {
   600→   600→    return <LoadingScreen />
   601→   601→  }
   602→   602→
   603→   603→  // Show fade-out loader + app content simultaneously
   604→   604→  return (
   605→   605→    <>
   606→   606→      {loaderFading && (
   607→   607→        <div id="initial-loader" className="fade-out">
   608→   608→          <div className="loader-spinner" />
   609→   609→          <p className="loader-text">
   610→   610→            Initializing Knowledge Base...
   611→   611→          </p>
   612→   612→        </div>
   613→   613→      )}
   614→   614→      <AppContent
   615→   615→          // Header
   616→   616→          searchQuery={searchQuery} onSearchInput={onSearchInput} searchInputRef={searchInputRef}
   617→   617→          toggleSidebar={toggleSidebar} onAddTerm={() => setAddModalOpen(true)}
   618→   618→          sidebarOpen={sidebarOpen} sidebarCollapsed={sidebarCollapsed} closeSidebar={closeSidebar}
   619→   619→          categories={categories} selectedCategory={selectedCategory} stats={stats}
   620→   620→          selectedFilter={selectedFilter} isDarkMode={isDarkMode} toggleDarkMode={toggleDarkMode}
   621→   621→          handleCategoryClick={handleCategoryClick} handleFilterClick={handleFilterClick}
   622→   622→          showArchGallery={showArchGallery} showPathsView={showPathsView}
   623→   623→          // Graph
   624→   624→          canvasRef={canvasRef} tooltipRef={tooltipRef} graphRef={graphRef}
   625→   625→          // Architecture
   626→   626→          activeView={activeView} archCurrentView={archCurrentView} currentArch={currentArch}
   627→   627→          showArchDetail={showArchDetail} hideArchView={hideArchView}
   628→   628→          architectures={architectures} setArchCurrentView={setArchCurrentView}
   629→   629→          // Knowledge paths
   630→   630→          pathMode={pathMode} activePath={activePath}
   631→   631→          pathSubPaths={pathSubPaths} knowledgePaths={knowledgePaths}
   632→   632→          startPath={startPath} focusTerm={focusTerm}
   633→   633→          setPathMode={setPathMode} setActivePath={setActivePath} setPathSubPaths={setPathSubPaths}
   634→   634→          setActiveView={setActiveView}
   635→   635→          // Detail panel
   636→   636→          detailPanelOpen={detailPanelOpen} selectedTerm={selectedTerm}
   637→   637→          selectedTermCategory={selectedTermCategory} relatedTerms={relatedTerms}
   638→   638→          copyState={copyState}
   639→   639→          onCopyCode={handleCopyCode}
   640→   640→          onRelatedClick={handleRelatedClick} onClosePanel={() => setDetailPanelOpen(false)}
   641→   641→          // Modal
   642→   642→          addModalOpen={addModalOpen} categoriesForModal={categories}
   643→   643→          handleAddTerm={handleAddTerm} onCloseModal={() => setAddModalOpen(false)}
   644→   644→          // Toast
   645→   645→          toast={toast}
   646→   646→          // Meta
   647→   647→          isMobile={isMobile}
   648→   648→        />
   649→   649→      </>
   650→   650→    )
