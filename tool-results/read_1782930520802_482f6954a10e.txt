   263→  loadData(categories: GraphCategory[], terms: GraphNode['term'][]) {
   264→    this.nodes = []
   265→    this.edges = []
   266→    this.nodeMap.clear()
   267→    this._labelCache.clear()
   268→    this._sortedNodes = []
   269→    this._sortedDirty = true
   270→
   271→    if (!categories.length || !terms.length) return
   272→
   273→    terms.forEach((term) => {
   274→      const category = categories.find((c) => c.id === term.category)
   275→      const categoryIndex = category ? categories.indexOf(category) : 0
   276→      const total = categories.length || 1
   277→
   278→      const baseAngle = (categoryIndex / total) * Math.PI * 2 - Math.PI / 2
   279→      const catTerms = terms.filter((t) => t.category === term.category)
   280→      const termIndex = catTerms.indexOf(term)
   281→
   282→      const radius = this.options.nodeRadius[term.type] ?? 16
   283→      const distance = term.type === 'core' ? 160 : term.type === 'technique' ? 220 : 280
   284→
   285→      const spreadAngle = Math.PI / 4
   286→      const angleOffset =
   287→        catTerms.length > 1
   288→          ? ((termIndex - (catTerms.length - 1) / 2) * spreadAngle) / catTerms.length
   289→          : 0
   290→      const angle = baseAngle + angleOffset
   291→
   292→      const jx = (Math.random() - 0.5) * 40
   293→      const jy = (Math.random() - 0.5) * 40
   294→
   295→      const node: GraphNode = {
   296→        id: term.id,
   297→        x: this.centerX + Math.cos(angle) * distance + jx,
   298→        y: this.centerY + Math.sin(angle) * distance + jy,
   299→        vx: 0,
   300→        vy: 0,
   301→        radius,
   302→        term,
   303→        color: category?.color ?? '#94a3b8',
   304→        highlighted: false,
   305→        visible: true,
   306→        currentRadius: radius,
   307→        targetRadius: radius,
   308→      }
   309→
   310→      this.nodes.push(node)
   311→      this.nodeMap.set(term.id, node)
   312→    })
   313→
   314→    const edgeSet = new Set<string>()
   315→    terms.forEach((term) => {
   316→      const relatedIds = (term as unknown as { related?: string[] }).related ?? []
   317→      relatedIds.forEach((relId: string) => {
   318→        const key = [term.id, relId].sort().join('|')
   319→        if (!edgeSet.has(key)) {
   320→          edgeSet.add(key)
   321→          this.edges.push({ source: term.id, target: relId, strength: 1 })
   322→        }
   323→      })
   324→    })
   325→
   326→    this._warmUpPhysics(120)
   327→  }

   329→  private _warmUpPhysics(iterations: number) {
   330→    const CHUNK = 60
   331→    let done = 0
   332→    const run = () => {
   333→      const end = Math.min(done + CHUNK, iterations)
   334→      for (; done < end; done++) this._simulatePhysics(0.15)
   335→      if (done < iterations) setTimeout(run, 0)
   336→    }
   337→    run()
   338→  }

   340→  private _simulatePhysics(dt = 1): boolean {
   341→    if (!this.physics.enabled) return false
   342→
   343→    const nodes = this.nodes
   344→    const CELL = 200
   345→    const eps = 0.001
   346→
   347→    const grid = new Map<string, GraphNode[]>()
   348→    const cellOf = (x: number, y: number) => `${Math.floor(x / CELL)},${Math.floor(y / CELL)}`
   349→
   350→    nodes.forEach((node) => {
   351→      if (!node.visible) return
   352→      const key = cellOf(node.x, node.y)
   353→      if (!grid.has(key)) grid.set(key, [])
   354→      grid.get(key)!.push(node)
   355→    })
   356→
   357→    const neighbourCells = (cx: number, cy: number) => {
   358→      const result: GraphNode[] = []
   359→      for (let dx = -1; dx <= 1; dx++) {
   360→        for (let dy = -1; dy <= 1; dy++) {
   361→          const key = `${cx + dx},${cy + dy}`
   362→          if (grid.has(key)) result.push(...grid.get(key)!)
   363→        }
   364→      }
   365→      return result
   366→    }
   367→
   368→    nodes.forEach((node) => {
   369→      if (!node.visible) return
   370→
   371→      node.vx += (this.centerX - node.x) * this.physics.centerGravity * dt
   372→      node.vy += (this.centerY - node.y) * this.physics.centerGravity * dt
   373→
   374→      const cx = Math.floor(node.x / CELL)
   375→      const cy = Math.floor(node.y / CELL)
   376→      const neighbours = neighbourCells(cx, cy)
   377→
   378→      neighbours.forEach((other) => {
   379→        if (node.id === other.id) return
   380→        const dx = node.x - other.x
   381→        const dy = node.y - other.y
   382→        const distSq = dx * dx + dy * dy
   383→        const dist = Math.max(Math.sqrt(distSq), 1)
   384→        if (dist < CELL) {
   385→          const force = this.physics.repulsion / (distSq + eps)
   386→          node.vx += (dx / dist) * force * dt
   387→          node.vy += (dy / dist) * force * dt
   388→        }
   389→      })
   390→    })
   391→
   392→    this.edges.forEach((edge) => {
   393→      const src = this.nodeMap.get(edge.source)
   394→      const tgt = this.nodeMap.get(edge.target)
   395→      if (!src || !tgt || !src.visible || !tgt.visible) return
   396→
   397→      const dx = tgt.x - src.x
   398→      const dy = tgt.y - src.y
   399→      const dist = Math.max(Math.sqrt(dx * dx + dy * dy), 1)
   400→      const targetDist = 180
   401→      const force = (dist - targetDist) * this.physics.attraction
   402→
   403→      src.vx += (dx / dist) * force * dt
   404→      src.vy += (dy / dist) * force * dt
   405→      tgt.vx -= (dx / dist) * force * dt
   406→      tgt.vy -= (dy / dist) * force * dt
   407→    })
   408→
   409→    const maxV = this.physics.maxVelocity
   410→    const padding = this.options.padding
   411→    const minV = this.physics.minVelocity
   412→    let moved = false
   413→
   414→    nodes.forEach((node) => {
   415→      if (!node.visible) return
   416→
   417→      node.vx *= this.physics.damping
   418→      node.vy *= this.physics.damping
   419→
   420→      const v = Math.sqrt(node.vx * node.vx + node.vy * node.vy)
   421→      if (v > maxV) {
   422→        node.vx = (node.vx / v) * maxV
   423→        node.vy = (node.vy / v) * maxV
   424→      }
   425→
   426→      if (Math.abs(node.vx) > minV) { node.x += node.vx; moved = true }
   427→      if (Math.abs(node.vy) > minV) { node.y += node.vy; moved = true }
   428→
   429→      node.x = Math.max(padding, Math.min(this.width - padding, node.x))
   430→      node.y = Math.max(padding, Math.min(this.height - padding, node.y))
   431→
   432→      const radiusDelta = (node.targetRadius - node.currentRadius) * 0.12
   433→      if (Math.abs(radiusDelta) > 0.01) {
   434→        node.currentRadius += radiusDelta
   435→        moved = true
   436→      }
   437→    })
   438→
   439→    return moved
   440→  }

   442→  screenToWorld(sx: number, sy: number) {
   443→    return { x: (sx - this.panX) / this.zoom, y: (sy - this.panY) / this.zoom }

   446→  private findNodeAt(wx: number, wy: number, extra = 0): GraphNode | null {
   447→    for (let i = this.nodes.length - 1; i >= 0; i--) {
   448→      const node = this.nodes[i]
   449→      if (!node.visible) continue
   450→      const dx = wx - node.x
   451→      const dy = wy - node.y
   452→      if (Math.sqrt(dx * dx + dy * dy) <= node.radius + 5 + extra) return node
   453→    }
   454→    return null
   455→  }

   457→  private _clientPos(e: MouseEvent) {

   462→  private _onMouseMove = (e: MouseEvent) => {
   463→    this._wake()
   464→    const { x, y } = this._clientPos(e)
   465→    if (this.isDragging) {
   466→      const dx = e.clientX - this.lastMouse.x
   467→      const dy = e.clientY - this.lastMouse.y
   468→      this.panX += dx
   469→      this.panY += dy
   470→      this._panTargetX = this.panX
   471→      this._panTargetY = this.panY
   472→      this.lastMouse = { x: e.clientX, y: e.clientY }
   473→    } else {
   474→      const world = this.screenToWorld(x, y)
   475→      const hovered = this.findNodeAt(world.x, world.y)
   476→      if (hovered !== this.hoveredNode) {
   477→        this.hoveredNode = hovered
   478→        this.canvas!.style.cursor = hovered ? 'pointer' : 'grab'
   479→        this._isDirty = true
   480→        if (this.onHoverChange) this.onHoverChange(hovered, e)
   481→      }
   482→    }
   483→  }

   488→    this._mouseDownPos = { x: e.clientX, y: e.clientY }
   489→    this.lastMouse = { x: e.clientX, y: e.clientY }
   490→    this.canvas!.style.cursor = 'grabbing'
   491→  }

   498→  private _onMouseLeave = () => {

   504→  private _onWheel = (e: WheelEvent) => {
   505→    this._wake()
   506→    e.preventDefault()
   507→    const { x, y } = this._clientPos(e)
   508→    const delta = e.deltaY > 0 ? 0.9 : 1.1
   509→    const newZoom = Math.max(this.options.zoomMin, Math.min(this.options.zoomMax, this.zoom * delta))
   510→    this._animateZoom(newZoom, x, y)
   511→  }
   512→
   513→  private _onClick = (e: MouseEvent) => {
   514→    if (e.clientX === undefined && e.clientY === undefined) return
   515→    // If mouse moved more than 5px from mousedown position, this was a drag/pan — skip click
   516→    if (this._mouseDownPos) {
   517→      const dx = e.clientX - this._mouseDownPos.x
   518→      const dy = e.clientY - this._mouseDownPos.y
   519→      if (Math.sqrt(dx * dx + dy * dy) > 5) return
   520→    }
   521→    const { x, y } = this._clientPos(e)
   522→    const world = this.screenToWorld(x, y)
   523→    const node = this.findNodeAt(world.x, world.y)
   524→
   525→    if (node) {
   526→      this.selectedNode = node
   527→      node.targetRadius = node.radius * 1.2
   528→      setTimeout(() => {
   529→        if (this.selectedNode === node) node.targetRadius = node.radius
   530→      }, 220)
   531→      if (this.onNodeSelect) this.onNodeSelect(node.term)
   532→    } else {
   533→      this.selectedNode = null
   534→    }
   535→  }

   537→  private _onTouchStart = (e: TouchEvent) => {
   538→    this._wake()
   539→    if (e.touches.length === 1) {
   540→      e.preventDefault()
   541→      const t = e.touches[0]
   542→      const rect = this.canvas!.getBoundingClientRect()
   543→      this._touchStartPos = { x: t.clientX - rect.left, y: t.clientY - rect.top }
   544→      this._touchStartTime = performance.now()
   545→      this.isDragging = true
   546→      this.lastMouse = { x: t.clientX, y: t.clientY }
   547→    } else if (e.touches.length === 2) {
   548→      e.preventDefault()
   549→      this.isDragging = false
   550→      const dx = e.touches[0].clientX - e.touches[1].clientX
   551→      const dy = e.touches[0].clientY - e.touches[1].clientY
   552→      this._pinchDist = Math.sqrt(dx * dx + dy * dy)
   553→      this._pinchZoom = this.zoom
   554→    }
   555→  }
   556→
   557→  private _onTouchMove = (e: TouchEvent) => {
   558→    if (e.touches.length === 1 && this.isDragging) {
   559→      e.preventDefault()
   560→      const t = e.touches[0]
   561→      const dx = t.clientX - this.lastMouse.x
   562→      const dy = t.clientY - this.lastMouse.y
   563→      this.panX += dx
   564→      this.panY += dy
   565→      this._panTargetX = this.panX
   566→      this._panTargetY = this.panY
   567→      this.lastMouse = { x: t.clientX, y: t.clientY }
   568→    } else if (e.touches.length === 2 && this._pinchDist) {
   569→      e.preventDefault()
   570→      const dx = e.touches[0].clientX - e.touches[1].clientX
   571→      const dy = e.touches[0].clientY - e.touches[1].clientY
   572→      const dist = Math.sqrt(dx * dx + dy * dy)
   573→      const factor = dist / this._pinchDist
   574→      const newZoom = Math.max(this.options.zoomMin, Math.min(this.options.zoomMax, this._pinchZoom * factor))
   575→      this._animateZoom(newZoom, this.width / 2, this.height / 2)
   576→    }
   577→  }
   578→
   579→  private _onTouchEnd = (e: TouchEvent) => {
   580→    if (e.changedTouches.length > 0 && this._touchStartPos) {
   581→      const t = e.changedTouches[0]
   582→      const rect = this.canvas!.getBoundingClientRect()
   583→      const ex = t.clientX - rect.left
   584→      const ey = t.clientY - rect.top
   585→
   586→      const distMoved = Math.sqrt((ex - this._touchStartPos.x) ** 2 + (ey - this._touchStartPos.y) ** 2)
   587→      const elapsed = performance.now() - this._touchStartTime
   588→
   589→      if (distMoved < this._TAP_MAX_DIST && elapsed < this._TAP_MAX_MS) {
   590→        const world = this.screenToWorld(ex, ey)
   591→        const node = this.findNodeAt(world.x, world.y, 12)
   592→        if (node) {
   593→          this.hoveredNode = node
   594→          this.selectedNode = node
   595→          node.targetRadius = node.radius * 1.2
   596→          setTimeout(() => {
   597→            if (this.selectedNode === node) node.targetRadius = node.radius
   598→          }, 220)
   599→          if (this.onNodeSelect) this.onNodeSelect(node.term)
   600→          if (this.onHoverChange) this.onHoverChange(node, e)
   601→        } else {
   602→          this.selectedNode = null
   603→          this.hoveredNode = null
   604→          if (this.onHoverChange) this.onHoverChange(null, null)
   605→        }
   606→      }
   607→    }
   608→
   609→    this.isDragging = false
   610→    this._pinchDist = null
   611→    this._touchStartPos = null
   612→    this.canvas!.style.cursor = 'grab'
   613→  }

   615→  private _bindEvents() {
   616→    if (!this.canvas) return
   617→    const add = (target: EventTarget, type: string, fn: EventListener, opts?: AddEventListenerOptions) => {
   618→      target.addEventListener(type, fn, opts)
   619→      this._boundListeners.push({ target, type, fn, opts })
   620→    }
   621→
   622→    add(window, 'resize', () => {
   623→      if (this._resizeTimer) clearTimeout(this._resizeTimer)
   624→      this._resizeTimer = setTimeout(() => this._handleResize(), 200)
   625→    })
   626→    add(this.canvas, 'mousemove', this._onMouseMove as EventListener)
   627→    add(this.canvas, 'mousedown', this._onMouseDown as EventListener)
   628→    add(this.canvas, 'mouseup', this._onMouseUp as EventListener)
   629→    add(this.canvas, 'mouseleave', this._onMouseLeave as EventListener)
   630→    add(this.canvas, 'wheel', this._onWheel as EventListener, { passive: false })
   631→    add(this.canvas, 'click', this._onClick as EventListener)
   632→    add(this.canvas, 'touchstart', this._onTouchStart as EventListener, { passive: false })
   633→    add(this.canvas, 'touchmove', this._onTouchMove as EventListener, { passive: false })
   634→    add(this.canvas, 'touchend', this._onTouchEnd as EventListener)
   635→
   636→    add(document, 'visibilitychange', () => {
   637→      if (document.hidden) {
   638→        this.stopAnimation()
   639→      } else {
   640→        this._lastTs = null
   641→        this._startAnimation()
   642→      }
   643→    })
   644→  }

   646→  private _handleResize() {
   647→    if (!this.canvas) return
   648→    const parent = this.canvas.parentElement
   649→    if (!parent) return
   650→    const rect = parent.getBoundingClientRect()
   651→
   652→    this.width = rect.width || 800
   653→    this.height = rect.height || 600
   654→    this.centerX = this.width / 2
   655→    this.centerY = this.height / 2
   656→    this.dpr = window.devicePixelRatio || 1
   657→
   658→    this.canvas.width = Math.round(this.width * this.dpr)
   659→    this.canvas.height = Math.round(this.height * this.dpr)
   660→    this.canvas.style.width = `${this.width}px`
   661→    this.canvas.style.height = `${this.height}px`
   662→
   663→    this._gridDirty = true
   664→    this._isDirty = true
   665→  }

   667→  private _animateZoom(newZoom: number, pivotX: number, pivotY: number) {
   668→    const scale = newZoom / this.zoom
   669→    this._panTargetX = pivotX - (pivotX - this.panX) * scale
   670→    this._panTargetY = pivotY - (pivotY - this.panY) * scale
   671→    this._zoomTarget = newZoom
   672→  }

   674→  private _startAnimation() {
   675→    if (this._isRunning) return
   676→
   677→    const tick = (ts: number) => {
   678→      const dt = this._lastTs ? Math.min((ts - this._lastTs) / 1000, 0.05) : 0.016
   679→      this._lastTs = ts
   680→
   681→      this.time += dt
   682→
   683→      const LERP = 0.18
   684→      const zooming = Math.abs(this._zoomTarget - this.zoom) > 0.0005
   685→      const panning = Math.abs(this._panTargetX - this.panX) > 0.1 || Math.abs(this._panTargetY - this.panY) > 0.1
   686→      if (zooming || panning) {
   687→        this.zoom += (this._zoomTarget - this.zoom) * LERP
   688→        this.panX += (this._panTargetX - this.panX) * LERP
   689→        this.panY += (this._panTargetY - this.panY) * LERP
   690→        this._isDirty = true
   691→      }
   692→
   693→      const hadMovement = this._simulatePhysics(dt * 60)
   694→      if (hadMovement) this._isDirty = true
   695→
   696→      // Skip render when idle (nothing changed for several frames)
   697→      const shouldRender = this._isDirty || this.hoveredNode || this.selectedNode || this._idleFrames < 3
   698→      if (shouldRender) {
   699→        this._render()
   700→        this._idleFrames = 0
   701→      } else {
   702→        this._idleFrames++
   703→      }
   704→
   705→      this._isDirty = false
   706→
   707→      // Full stop when truly idle — saves CPU/battery
   708→      if (this._idleFrames >= this._MAX_IDLE) {
   709→        this._isRunning = false
   710→        this.animationId = null
   711→        return
   712→      }
   713→
   714→      this.animationId = requestAnimationFrame(tick)
   715→    }
   716→
   717→    this._isRunning = true
   718→    this.animationId = requestAnimationFrame(tick)
   719→  }

   721→  private _rebuildGridCache() {
   722→    const w = this.width
   723→    const h = this.height
   724→    const cw = Math.round(w * this.dpr)
   725→    const ch = Math.round(h * this.dpr)
   726→
   727→    // Reuse existing canvas if dimensions haven't changed
   728→    if (!this._gridCanvas) {
   729→      this._gridCanvas = document.createElement('canvas')
   730→      this._gridCanvas.width = cw
   731→      this._gridCanvas.height = ch
   732→    } else if (this._gridCanvas.width !== cw || this._gridCanvas.height !== ch) {
   733→      this._gridCanvas.width = cw
   734→      this._gridCanvas.height = ch
   735→    }
   736→    // else: dimensions match, just repaint
   737→
   738→    const gc = this._gridCanvas.getContext('2d')!
   739→    gc.setTransform(this.dpr, 0, 0, this.dpr, 0, 0)
   740→    gc.fillStyle = this.theme.gridDot
   741→
   742→    const gap = 40
   743→    for (let x = gap; x < w; x += gap) {
   744→      for (let y = gap; y < h; y += gap) {
   745→        gc.beginPath()
   746→        gc.arc(x, y, 1.5, 0, Math.PI * 2)
   747→        gc.fill()
   748→      }
   749→    }
   750→
   751→    this._gridDirty = false
   752→  }
   753→
   754→  private _render() {
   755→    const ctx = this.ctx
   756→    if (!ctx || !this.canvas) return
   757→
   758→    // Track selection changes for sort optimization
   759→    const curSelId = this.selectedNode?.id ?? null
   760→    if (curSelId !== this._prevSelectedId) {
   761→      this._sortedDirty = true
   762→      this._prevSelectedId = curSelId
   763→    }
   764→
   765→    ctx.setTransform(this.dpr, 0, 0, this.dpr, 0, 0)
   766→
   767→    const bgGrad = ctx.createRadialGradient(this.centerX, this.centerY, 0, this.centerX, this.centerY, Math.max(this.width, this.height) * 0.7)
   768→    bgGrad.addColorStop(0, this.theme.bgGradientCenter)
   769→    bgGrad.addColorStop(1, this.theme.bg)
   770→    ctx.fillStyle = bgGrad
   771→    ctx.fillRect(0, 0, this.width, this.height)
   772→
   773→    if (this._gridDirty || !this._gridCanvas) this._rebuildGridCache()
   774→    if (this._gridCanvas) ctx.drawImage(this._gridCanvas, 0, 0, this.width, this.height)
   775→
   776→    ctx.save()
   777→    ctx.translate(this.panX, this.panY)
   778→    ctx.scale(this.zoom, this.zoom)
   779→
   780→    this._drawEdges(ctx)
   781→    this._drawNodes(ctx)
   782→
   783→    ctx.restore()
   784→  }
   785→
   786→  private _drawEdges(ctx: CanvasRenderingContext2D) {
   787→    const selected = this.selectedNode
   788→    const hovered = this.hoveredNode
   789→
   790→    this.edges.forEach((edge) => {
   791→      const src = this.nodeMap.get(edge.source)
   792→      const tgt = this.nodeMap.get(edge.target)
   793→      if (!src || !tgt || !src.visible || !tgt.visible) return
   794→
   795→      const srcActive = selected && (selected.id === src.id || selected.id === tgt.id)
   796→      const hovActive = hovered && (hovered.id === src.id || hovered.id === tgt.id)
   797→
   798→      let opacity: number
   799→      if (selected) {
   800→        opacity = srcActive ? 1.0 : 0.06
   801→      } else if (hovered) {
   802→        opacity = hovActive ? 0.85 : 0.12
   803→      } else {
   804→        opacity = 0.55
   805→      }
   806→
   807→      const dx = tgt.x - src.x
   808→      const dy = tgt.y - src.y
   809→      const dist = Math.max(Math.sqrt(dx * dx + dy * dy), 1)
   810→
   811→      const mx = (src.x + tgt.x) / 2
   812→      const my = (src.y + tgt.y) / 2
   813→      // Edge curve direction: since edges are bidirectional (no inherent direction),
   814→      // we use a deterministic but arbitrary ID-based comparison so that the curve bends
   815→      // consistently for each pair rather than randomly flipping.
   816→      const off = dist * 0.15
   817→      const dir = src.id < tgt.id ? 1 : -1
   818→      const cx = mx + (dy / dist) * off * dir
   819→      const cy = my - (dx / dist) * off * dir
   820→
   821→      const grad = ctx.createLinearGradient(src.x, src.y, tgt.x, tgt.y)
   822→      grad.addColorStop(0, this._hexToRgba(src.color, opacity))
   823→      grad.addColorStop(1, this._hexToRgba(tgt.color, opacity))
   824→
   825→      ctx.beginPath()
   826→      ctx.moveTo(src.x, src.y)
   827→      ctx.quadraticCurveTo(cx, cy, tgt.x, tgt.y)
   828→      ctx.strokeStyle = grad
   829→      ctx.lineWidth = srcActive || hovActive ? 2.5 : 1.5
   830→      ctx.lineCap = 'round'
   831→      ctx.setLineDash([])
   832→      ctx.stroke()
   833→
   834→      // Arrowhead
   835→      const tangX = tgt.x - cx
   836→      const tangY = tgt.y - cy
   837→      const angle = Math.atan2(tangY, tangX)
   838→      const arrowX = tgt.x - Math.cos(angle) * (tgt.radius + 6)
   839→      const arrowY = tgt.y - Math.sin(angle) * (tgt.radius + 6)
   840→
   841→      ctx.save()
   842→      ctx.translate(arrowX, arrowY)
   843→      ctx.rotate(angle)
   844→      ctx.beginPath()
   845→      ctx.moveTo(0, 0)
   846→      ctx.lineTo(-8, 4)
   847→      ctx.lineTo(-8, -4)
   848→      ctx.closePath()
   849→      ctx.fillStyle = this._hexToRgba(tgt.color, opacity)
   850→      ctx.fill()
   851→      ctx.restore()
   852→
   853→      // Quantum packets on active edges
   854→      if (srcActive || hovActive) {
   855→        ctx.save()
   856→        ctx.shadowBlur = 12
   857→        ctx.shadowColor = this._isDark ? '#60a5fa' : '#ffffff'
   858→
   859→        const packetColor = this._isDark ? '#93c5fd' : '#ffffff'
   860→        for (let i = 0; i < 3; i++) {
   861→          const t = ((this.time * 0.6 + i / 3)) % 1
   862→          const pos = this._bezierPoint(t, src.x, src.y, cx, cy, tgt.x, tgt.y)
   863→          const sz = 2 + Math.sin(this.time * 10 + i) * 0.5
   864→
   865→          ctx.beginPath()
   866→          ctx.arc(pos.x, pos.y, sz, 0, Math.PI * 2)
   867→          ctx.fillStyle = packetColor
   868→          ctx.fill()
   869→        }
   870→
   871→        ctx.restore()
   872→      }
   873→    })
   874→  }

   876→  private _drawNodes(ctx: CanvasRenderingContext2D) {
   877→    // Re-sort when selection changes — separate sorting from drawing to avoid mutation
   878→    if (this._sortedDirty) {
   879→      this._sortedNodes = this.nodes.slice()
   880→      // Move selected node to end for draw order (on top)
   881→      const selId = this.selectedNode?.id
   882→      if (selId) {
   883→        const selIdx = this._sortedNodes.findIndex((n) => n.id === selId)
   884→        if (selIdx !== -1 && selIdx !== this._sortedNodes.length - 1) {
   885→          const [selNode] = this._sortedNodes.splice(selIdx, 1)
   886→          this._sortedNodes.push(selNode)
   887→        }
   888→      }
   889→      this._sortedDirty = false
   890→    }
   891→
   892→    const nodes = this._sortedNodes
   893→    const selId = this.selectedNode?.id
   894→    const hovId = this.hoveredNode?.id
   895→    const hasSelection = !!selId
   896→    const fontSize = this.options.fontSize
   897→    const fontFamily = "'Plus Jakarta Sans', sans-serif"
   898→    const time = this.time
   899→    const isDark = this._isDark
   900→
   901→    for (let ni = 0; ni < nodes.length; ni++) {
   902→      const node = nodes[ni]
   903→      if (!node.visible) continue
   904→
   905→      const isSel = selId === node.id
   906→      const isHov = hovId === node.id
   907→      const dimmed = hasSelection && !isSel && !node.highlighted
   908→
   909→      let r = Math.max(1, node.currentRadius + (isSel ? Math.sin(time * 4) * 1.5 : 0))
   910→
   911→      // Outer scanning rings (selected)
   912→      if (isSel) {
   913→        ctx.save()
   914→        ctx.translate(node.x, node.y)
   915→        ctx.globalAlpha = 0.6
   916→        ctx.strokeStyle = node.color
   917→        ctx.lineWidth = 1.5
   918→        ctx.setLineDash([4, 8])
   919→        ctx.rotate(-time)
   920→        ctx.beginPath()
   921→        ctx.arc(0, 0, r + 8, 0, Math.PI * 2)
   922→        ctx.stroke()
   923→
   924→        ctx.setLineDash([])
   925→        ctx.lineWidth = 1
   926→        ctx.globalAlpha = 0.3 + Math.sin(time * 5) * 0.2
   927→        ctx.rotate(time * 2)
   928→        ctx.beginPath()
   929→        ctx.arc(0, 0, r + 4 + Math.sin(time * 3) * 3, 0, Math.PI * 2)
   930→        ctx.stroke()
   931→        ctx.restore()
   932→      }
   933→
   934→      // Highlighted ring (search result)
   935→      if (node.highlighted && !isSel) {
   936→        ctx.save()
   937→        ctx.globalAlpha = 0.7 + Math.sin(time * 6) * 0.2
   938→        ctx.strokeStyle = node.color
   939→        ctx.lineWidth = 2
   940→        ctx.setLineDash([3, 5])
   941→        ctx.beginPath()
   942→        ctx.arc(node.x, node.y, r + 5, 0, Math.PI * 2)
   943→        ctx.stroke()
   944→        ctx.setLineDash([])
   945→        ctx.restore()
   946→      }
   947→
   948→      // Shadow / glow
   949→      ctx.shadowColor = isSel || isHov ? node.color : 'rgba(0,0,0,0.08)'
   950→      ctx.shadowBlur = isSel || isHov ? 22 : 5
   951→
   952→      // Node body
   953→      ctx.beginPath()
   954→      ctx.arc(node.x, node.y, r, 0, Math.PI * 2)
   955→
   956→      const grad = ctx.createRadialGradient(node.x - r * 0.3, node.y - r * 0.3, r * 0.1, node.x, node.y, r)
   957→
   958→      if (isSel || isHov || node.highlighted) {
   959→        grad.addColorStop(0, isDark ? '#334155' : '#ffffff')
   960→        grad.addColorStop(0.4, this._hexToRgba(node.color, dimmed ? 0.3 : 1))
   961→        grad.addColorStop(1, this._darkenColor(node.color, 20))
   962→      } else {
   963→        grad.addColorStop(0, dimmed ? this.theme.dimmedBodyStart : this.theme.nodeBodyStart)
   964→        grad.addColorStop(1, dimmed ? this.theme.dimmedBodyEnd : this.theme.nodeBodyEnd)
   965→      }
   966→
   967→      ctx.globalAlpha = dimmed ? 0.35 : 1
   968→      ctx.fillStyle = grad
   969→      ctx.fill()
   970→      ctx.globalAlpha = 1
   971→      ctx.shadowBlur = 0
   972→
   973→      // Border
   974→      ctx.strokeStyle = isSel || isHov ? node.color : this.theme.nodeBorderDefault
   975→      ctx.lineWidth = isSel || isHov ? 2.5 : 1
   976→      ctx.stroke()
   977→
   978→      // Label (with cache)
   979→      const maxLabelPx = Math.round((r - 3) * 2)
   980→      const cacheKey = `${node.id}:${maxLabelPx}`
   981→      let label = this._labelCache.get(cacheKey)
   982→      if (label === undefined) {
   983→        ctx.font = `600 ${fontSize}px ${fontFamily}`
   984→        label = this._truncateLabel(ctx, node.term.name, maxLabelPx)
   985→        this._labelCache.set(cacheKey, label)
   986→      }
   987→      ctx.font = `600 ${fontSize}px ${fontFamily}`
   988→      ctx.textAlign = 'center'
   989→      ctx.textBaseline = 'middle'
   990→      ctx.fillStyle = isSel ? (isDark ? '#e8f0fe' : '#0f172a') : isHov ? node.color : this.theme.nodeLabelDefault
   991→
   992→      ctx.shadowColor = 'rgba(255,255,255,0.85)'
   993→      ctx.shadowBlur = 4
   994→      ctx.fillText(label, node.x, node.y)
   995→      ctx.shadowBlur = 0
   996→    }
   997→  }

   999→  private _prevSelectedId: string | null = null

  1001→  private _truncateLabel(ctx: CanvasRenderingContext2D, text: string, maxPx: number): string {
  1002→    if (ctx.measureText(text).width <= maxPx) return text
  1003→    let t = text
  1004→    while (t.length > 1 && ctx.measureText(t + '\u2026').width > maxPx) {
  1005→      t = t.slice(0, -1)
  1006→    }
  1007→    return t + '\u2026'
  1008→  }
  1009→
  1010→  private _bezierPoint(t: number, sx: number, sy: number, cx: number, cy: number, tx: number, ty: number) {
  1011→    const u = 1 - t
  1012→    return {
  1013→      x: u * u * sx + 2 * u * t * cx + t * t * tx,
  1014→      y: u * u * sy + 2 * u * t * cy + t * t * ty,
  1015→    }
  1016→  }
  1017→
  1018→  private _hexToRgba(hex: string, alpha = 1): string {
  1019→    if (typeof hex !== 'string' || !hex.startsWith('#')) return `rgba(148,163,184,${alpha})`
  1020→
  1021→    let h = hex.slice(1)
  1022→    if (h.length === 3) h = h[0] + h[0] + h[1] + h[1] + h[2] + h[2]
  1023→    if (h.length !== 6) return `rgba(148,163,184,${alpha})`
  1024→
  1025→    const r = parseInt(h.slice(0, 2), 16)
  1026→    const g = parseInt(h.slice(2, 4), 16)
  1027→    const b = parseInt(h.slice(4, 6), 16)
  1028→
  1029→    if (isNaN(r) || isNaN(g) || isNaN(b)) return `rgba(148,163,184,${alpha})`
  1030→    return `rgba(${r},${g},${b},${alpha})`
  1031→  }
  1032→
  1033→  private _darkenColor(hex: string, percent: number): string {
  1034→    if (typeof hex !== 'string' || !hex.startsWith('#')) return '#94a3b8'
  1035→
  1036→    let h = hex.slice(1)
  1037→    if (h.length === 3) h = h[0] + h[0] + h[1] + h[1] + h[2] + h[2]
  1038→    if (h.length !== 6) return '#94a3b8'
  1039→
  1040→    const amt = Math.round(2.55 * percent)
  1041→    const num = parseInt(h, 16)
  1042→    const R = Math.max(0, (num >> 16) - amt)
  1043→    const G = Math.max(0, ((num >> 8) & 0x00ff) - amt)
  1044→    const B = Math.max(0, (num & 0xff) - amt)
  1045→    return '#' + (0x1000000 + R * 0x10000 + G * 0x100 + B).toString(16).slice(1)
  1046→  }
