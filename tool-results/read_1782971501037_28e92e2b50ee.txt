     8→     8→function escapeHTML(text: string): string {
     9→     9→  return String(text ?? '')
    10→    10→    .replace(/&/g, '&amp;')
    11→    11→    .replace(/</g, '&lt;')
    12→    12→    .replace(/>/g, '&gt;')
    13→    13→    .replace(/"/g, '&quot;')
    14→    14→    .replace(/'/g, '&#39;')
    15→    15→}

    17→    17→function parseMarkdown(text: string): string {
    18→    18→  if (!text) return ''
    19→    19→  let out = escapeHTML(text)
    20→    20→  out = out.replace(/```([\s\S]*?)```/g, (_, code) => `<pre><code>${code.trim()}</code></pre>`)
    21→    21→  out = out.replace(/`([^`\n]+)`/g, '<code>$1</code>')
    22→    22→  out = out.replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>')
    23→    23→  out = out.replace(/(?<!\*)\*(?!\*)(.+?)(?<!\*)\*(?!\*)/g, '<em>$1</em>')
    24→    24→  out = out.replace(/^### (.+)$/gm, '<h4>$1</h4>')
    25→    25→  out = out.replace(/^## (.+)$/gm, '<h3>$1</h3>')
    26→    26→  out = out.replace(/^# (.+)$/gm, '<h2>$1</h2>')
    27→    27→  out = out.replace(
    28→    28→    /(?:^- .+\n?)+/gm,
    29→    29→    (block) =>
    30→    30→      '<ul>' +
    31→    31→      block
    32→    32→        .trim()
    33→    33→        .split('\n')
    34→    34→        .map((l) => `<li>${l.replace(/^- /, '').trim()}</li>`)
    35→    35→        .join('') +
    36→    36→      '</ul>'
    37→    37→  )
    38→    38→  out = out.replace(
    39→    39→    /(?:^\d+\. .+\n?)+/gm,
    40→    40→    (block) =>
    41→    41→      '<ol>' +
    42→    42→      block
    43→    43→        .trim()
    44→    44→        .split('\n')
    45→    45→        .map((l) => `<li>${l.replace(/^\d+\. /, '').trim()}</li>`)
    46→    46→        .join('') +
    47→    47→      '</ol>'
    48→    48→  )
    49→    49→  const BLOCK_START = /^<(ul|ol|pre|h[1-6]|blockquote)/
    50→    50→  out = out
    51→    51→    .split(/\n{2,}/)
    52→    52→    .map((block) => {
    53→    53→      const trimmed = block.trim()
    54→    54→      if (!trimmed) return ''
    55→    55→      if (BLOCK_START.test(trimmed)) return trimmed
    56→    56→      return `<p>${trimmed.replace(/\n/g, '<br>')}</p>`
    57→    57→    })
    58→    58→    .filter(Boolean)
    59→    59→    .join('')
    60→    60→  return out
    61→    61→}

    63→    63→// ─── Architecture step renderer ───
    64→    64→function renderArchSteps(steps: { label: string; desc?: string; type?: string; inputId?: string; children?: typeof steps }[]): string {
    65→    65→  if (!steps || !steps.length) return ''
    66→    66→  return steps
    67→    67→    .map((step) => {
    68→    68→      let typeClass = `step-type-${step.type || 'process'}`
    69→    69→      let childrenHtml = ''
    70→    70→      if (step.children && step.children.length > 0) {
    71→    71→        typeClass += ' step-type-container'
    72→    72→        childrenHtml = `<div class="step-children">${renderArchSteps(step.children)}</div>`
    73→    73→      }
    74→    74→      return `
    75→    75→        <div class="flow-step">
    76→    76→          <div class="step-box ${typeClass}">
    77→    77→            <div class="step-label">${escapeHTML(step.label)}</div>
    78→    78→            ${step.desc ? `<div class="step-desc">${escapeHTML(step.desc)}</div>` : ''}
    79→    79→          </div>
    80→    80→          ${childrenHtml}
    81→    81→        </div>
    82→    82→      `
    83→    83→    })
    84→    84→    .join('')
    85→    85→}

   117→   117→  const bg =
   118→   118→    toast.type === 'success'
   119→   119→      ? 'var(--accent-success, #10b981)'
   120→   120→      : toast.type === 'error'
   121→   121→        ? 'var(--accent-danger, #ef4444)'
   122→   122→        : 'var(--accent-primary, #0891b2)'
   123→   123→
   124→   124→  return (
   125→   125→    <div
   126→   126→      key={toast.key}
   127→   127→      className={`app-toast ${toast.exiting ? 'toast-exit' : ''}`}
   128→   128→      style={{ background: bg }}
   129→   129→      role="status"
   130→   130→      aria-live="polite"
   131→   131→    >
   132→   132→      {toast.message}
   133→   133→    </div>
   134→   134→  )

   439→   439→  const [copyState, setCopyState] = useState<'idle' | 'copied' | 'failed'>('idle')
   440→   440→
   441→   441→  const handleCopyCode = useCallback(async () => {
   442→   442→    const code = selectedTerm?.codeExample
   443→   443→    if (!code) return
   444→   444→    try {
   445→   445→      if (navigator.clipboard?.writeText) {
   446→   446→        await navigator.clipboard.writeText(code)
   447→   447→      } else {
   448→   448→        const ta = document.createElement('textarea')
   449→   449→        ta.value = code
   450→   450→        ta.style.cssText = 'position:fixed;top:-999px;left:-999px;opacity:0;'
   451→   451→        document.body.appendChild(ta)
   452→   452→        ta.select()
   453→   453→        document.execCommand('copy')
   454→   454→        ta.remove()
   455→   455→      }
   456→   456→      setCopyState('copied')
   457→   457→      setTimeout(() => setCopyState('idle'), 1500)
   458→   458→    } catch {
   459→   459→      setCopyState('failed')
   460→   460→      setTimeout(() => setCopyState('idle'), 1500)
   461→   461→    }
   462→   462→  }, [selectedTerm?.codeExample])
   463→   463→
   464→   464→  // Reset copy state when term changes
   465→   465→  useEffect(() => {
   466→   466→    setCopyState('idle')
   467→   467→  }, [selectedTerm])
   468→   468→
   469→   469→  // ─── Handlers ───
   470→   470→  const handleCategoryClick = useCallback((categoryId: string) => {
   471→   471→    if (selectedCategory === categoryId) {
   472→   472→      setSelectedCategory(null)
   473→   473→      graphRef.current?.filterByCategory(null)
   474→   474→    } else {
   475→   475→      setSelectedCategory(categoryId)
   476→   476→      graphRef.current?.filterByCategory(categoryId)
   477→   477→    }
   478→   478→  }, [selectedCategory, setSelectedCategory])
   479→   479→
   480→   480→  const handleFilterClick = useCallback((filter: string) => {
   481→   481→    setSelectedFilter(filter)
   482→   482→    graphRef.current?.filterByType(filter === 'all' ? null : filter)
   483→   483→    if (filter === 'all') setSelectedCategory(null)
   484→   484→  }, [setSelectedFilter, setSelectedCategory])
   485→   485→
   486→   486→  const handleRelatedClick = useCallback((termId: string) => {
   487→   487→    const term = terms.find((t) => t.id === termId)
   488→   488→    if (term) {
   489→   489→      setSelectedTerm(term)
   490→   490→      navigateTerm(term.id)
   491→   491→      if (graphRef.current) {
   492→   492→        graphRef.current.selectedNode = graphRef.current.findNode(term.id)
   493→   493→      }
   494→   494→    }
   495→   495→  }, [terms, setSelectedTerm, navigateTerm])
   496→   496→
   497→   497→  const handleAddTerm = useCallback((e: React.FormEvent<HTMLFormElement>) => {
   498→   498→    e.preventDefault()
   499→   499→    const formData = new FormData(e.currentTarget)
   500→   500→    const name = formData.get('name')?.toString().trim() ?? ''
   501→   501→    const categoryId = formData.get('category')?.toString().trim() ?? ''
   502→   502→    const shortDesc = formData.get('shortDesc')?.toString().trim() ?? ''
   503→   503→    const definition = formData.get('definition')?.toString().trim() ?? ''
   504→   504→    const relatedStr = formData.get('related')?.toString().trim() ?? ''
   505→   505→    const tagsStr = formData.get('tags')?.toString().trim() ?? ''
   506→   506→
   507→   507→    if (!name || !categoryId || !shortDesc || !definition) {
   508→   508→      showToast('Please fill all required fields.', 'error')
   509→   509→      return
   510→   510→    }
   511→   511→
   512→   512→    const baseId = name.toLowerCase().replace(/\s+/g, '-').replace(/[^a-z0-9-]/g, '')
   513→   513→    if (!baseId) {
   514→   514→      showToast('Could not generate a valid ID from that name.', 'error')
   515→   515→      return
   516→   516→    }
   517→   517→
   518→   518→    let id = baseId
   519→   519→    let suffix = 1
   520→   520→    while (terms.find((t) => t.id === id)) {
   521→   521→      id = `${baseId}-${++suffix}`
   522→   522→    }
   523→   523→
   524→   524→    const related = relatedStr
   525→   525→      .split(',')
   526→   526→      .map((s) => s.trim().toLowerCase().replace(/\s+/g, '-').replace(/[^a-z0-9-]/g, ''))
   527→   527→      .filter(Boolean)
   528→   528→
   529→   529→    const tags = tagsStr
   530→   530→      .split(',')
   531→   531→      .map((s) => s.trim())
   532→   532→      .filter(Boolean)
   533→   533→
   534→   534→    addTerm({
   535→   535→      id, name, fullName: name, category: categoryId, type: 'technique',
   536→   536→      shortDesc, definition, related, tags, codeExample: '',
   537→   537→      createdAt: new Date().toISOString(), importance: 0,
   538→   538→    })
   539→   539→
   540→   540→    if (graphRef.current) {
   541→   541→      const cats = categories.map((c) => ({ id: c.id, name: c.name, color: c.color }))
   542→   542→      const updatedTerms = useTitanMLStore.getState().terms
   543→   543→      const termsData = updatedTerms.map((t) => ({
   544→   544→        id: t.id, name: t.name, fullName: t.fullName,
   545→   545→        category: t.category, type: t.type, shortDesc: t.shortDesc,
   546→   546→        tags: t.tags, related: t.related,
   547→   547→      }))
   548→   548→      graphRef.current.loadData(cats, termsData)
   549→   549→    }
   550→   550→
   551→   551→    setAddModalOpen(false)
   552→   552→    showToast(`"${name}" added successfully!`, 'success')
   553→   553→  }, [addTerm, categories, terms, setAddModalOpen, showToast])
   554→   554→
   555→   555→  // Architecture handlers
   556→   556→  const showArchGallery = useCallback(() => {
   557→   557→    setActiveView('architecture')
   558→   558→    setArchCurrentView('gallery')
   559→   559→    setArchCurrentId(null)
   560→   560→  }, [setActiveView, setArchCurrentView, setArchCurrentId])
   561→   561→
   562→   562→  const showArchDetail = useCallback((archId: string) => {
   563→   563→    setArchCurrentId(archId)
   564→   564→    setArchCurrentView('architecture')
   565→   565→  }, [setArchCurrentId, setArchCurrentView])
   566→   566→
   567→   567→  const hideArchView = useCallback(() => {
   568→   568→    setActiveView('graph')
   569→   569→    setArchCurrentId(null)
   570→   570→  }, [setActiveView, setArchCurrentId])
   571→   571→
   572→   572→  // Knowledge path handlers
   573→   573→  const showPathsView = useCallback(() => {
   574→   574→    setActiveView('knowledge-paths')
   575→   575→    setPathMode('gallery')
   576→   576→    setActivePath(null)
   577→   577→    setPathSubPaths([])
   578→   578→  }, [setActiveView, setPathMode, setActivePath, setPathSubPaths])
   579→   579→
   580→   580→  const startPath = useCallback((path: KnowledgePath) => {
   581→   581→    if (path.paths && !path.steps) {
   582→   582→      setPathMode('subpaths')
   583→   583→      setPathSubPaths(path.paths)
   584→   584→      setActivePath(path)
   585→   585→    } else {
   586→   586→      setPathMode('steps')
   587→   587→      setActivePath(path)
   588→   588→    }
   589→   589→  }, [setPathMode, setPathSubPaths, setActivePath])
   590→   590→
   591→   591→  const focusTerm = useCallback((termName: string) => {
   592→   592→    setActiveView('graph')
   593→   593→    const term = terms.find((t) => t.name === termName || t.fullName === termName)
   594→   594→    if (term) {
   595→   595→      setSelectedTerm(term)
   596→   596→      navigateTerm(term.id)
   597→   597→    }
   598→   598→  }, [terms, setActiveView, setSelectedTerm, navigateTerm])
   599→   599→
   600→   600→  // ─── Memoized derived data ───
   601→   601→  const selectedTermCategory = useMemo(() => {
   602→   602→    if (!selectedTerm) return null
   603→   603→    return categories.find((c) => c.id === selectedTerm.category) ?? null
   604→   604→  }, [selectedTerm, categories])
   605→   605→
   606→   606→  const relatedTerms = useMemo(() => {
   607→   607→    if (!selectedTerm) return []
   608→   608→    const results: { term: Term; cat: Category | undefined }[] = []
   609→   609→    for (let i = 0; i < selectedTerm.related.length; i++) {
   610→   610→      const rt = terms.find((t) => t.id === selectedTerm.related[i])
   611→   611→      if (!rt) continue
   612→   612→      const rc = categories.find((c) => c.id === rt.category)
   613→   613→      results.push({ term: rt, cat: rc })
   614→   614→    }
   615→   615→    return results
   616→   616→  }, [selectedTerm, terms, categories])
   617→   617→
   618→   618→  const currentArch = useMemo(() => {
   619→   619→    if (!archCurrentId) return null
   620→   620→    return architectures.find((a) => a.id === archCurrentId) ?? null
   621→   621→  }, [archCurrentId, architectures])
   622→   622→
   623→   623→  // ─── Loading screen ───
   624→   624→  if (!isLoaded) {
   625→   625→    return <LoadingScreen />
   626→   626→  }
   627→   627→
   628→   628→  // Show fade-out loader + app content simultaneously
   629→   629→  return (
   630→   630→    <>
   631→   631→      {loaderFading && (
   632→   632→        <div id="initial-loader" className="fade-out">
   633→   633→          <div className="loader-spinner" />
   634→   634→          <p className="loader-text">
   635→   635→            Initializing Knowledge Base...
   636→   636→          </p>
   637→   637→        </div>
   638→   638→      )}
   639→   639→      <AppContent
   640→   640→          // Header
   641→   641→          searchQuery={searchQuery} onSearchInput={onSearchInput} searchInputRef={searchInputRef}
   642→   642→          toggleSidebar={toggleSidebar} onAddTerm={() => setAddModalOpen(true)}
   643→   643→          sidebarOpen={sidebarOpen} sidebarCollapsed={sidebarCollapsed} closeSidebar={closeSidebar}
   644→   644→          categories={categories} selectedCategory={selectedCategory} stats={stats}
   645→   645→          selectedFilter={selectedFilter} isDarkMode={isDarkMode} toggleDarkMode={toggleDarkMode}
   646→   646→          handleCategoryClick={handleCategoryClick} handleFilterClick={handleFilterClick}
   647→   647→          showArchGallery={showArchGallery} showPathsView={showPathsView}
   648→   648→          // Graph
   649→   649→          canvasRef={canvasRef} tooltipRef={tooltipRef} graphRef={graphRef}
   650→   650→          // Architecture
   651→   651→          activeView={activeView} archCurrentView={archCurrentView} currentArch={currentArch}
   652→   652→          showArchDetail={showArchDetail} hideArchView={hideArchView}
   653→   653→          architectures={architectures} setArchCurrentView={setArchCurrentView}
   654→   654→          // Knowledge paths
   655→   655→          pathMode={pathMode} activePath={activePath}
   656→   656→          pathSubPaths={pathSubPaths} knowledgePaths={knowledgePaths}
   657→   657→          startPath={startPath} focusTerm={focusTerm}
   658→   658→          setPathMode={setPathMode} setActivePath={setActivePath} setPathSubPaths={setPathSubPaths}
   659→   659→          setActiveView={setActiveView}
   660→   660→          // Terms (for path step existence check)
   661→   661→          terms={terms}
   662→   662→          // Detail panel
   663→   663→          detailPanelOpen={detailPanelOpen} selectedTerm={selectedTerm}
   664→   664→          selectedTermCategory={selectedTermCategory} relatedTerms={relatedTerms}
   665→   665→          copyState={copyState}
   666→   666→          onCopyCode={handleCopyCode}
   667→   667→          onRelatedClick={handleRelatedClick} onClosePanel={() => setDetailPanelOpen(false)}
   668→   668→          // Modal
   669→   669→          addModalOpen={addModalOpen} categoriesForModal={categories}
   670→   670→          handleAddTerm={handleAddTerm} onCloseModal={() => setAddModalOpen(false)}
   671→   671→          // Toast
   672→   672→          toast={toast}
   673→   673→          // Meta
   674→   674→          isMobile={isMobile}
   675→   675→        />
   676→   676→      </>
   677→   677→    )
   678→   678→}
