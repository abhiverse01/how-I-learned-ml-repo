     7→// ─── Utility functions ───
     8→function escapeHTML(text: string): string {
     9→  return String(text ?? '')
    10→    .replace(/&/g, '&amp;')
    11→    .replace(/</g, '&lt;')
    12→    .replace(/>/g, '&gt;')
    13→    .replace(/"/g, '&quot;')
    14→    .replace(/'/g, '&#39;')
    15→}
    16→
    17→function parseMarkdown(text: string): string {
    18→  if (!text) return ''
    19→  let out = escapeHTML(text)
    20→  out = out.replace(/```([\s\S]*?)```/g, (_, code) => `<pre><code>${code.trim()}</code></pre>`)
    21→  out = out.replace(/`([^`\n]+)`/g, '<code>$1</code>')
    22→  out = out.replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>')
    23→  out = out.replace(/(?<!\*)\*(?!\*)(.+?)(?<!\*)\*(?!\*)/g, '<em>$1</em>')
    24→  out = out.replace(/^### (.+)$/gm, '<h4>$1</h4>')
    25→  out = out.replace(/^## (.+)$/gm, '<h3>$1</h3>')
    26→  out = out.replace(/^# (.+)$/gm, '<h2>$1</h2>')
    27→  out = out.replace(
    28→    /(?:^- .+\n?)+/gm,
    29→    (block) =>
    30→      '<ul>' +
    31→      block
    32→        .trim()
    33→        .split('\n')
    34→        .map((l) => `<li>${l.replace(/^- /, '').trim()}</li>`)
    35→        .join('') +
    36→      '</ul>'
    37→  )
    38→  out = out.replace(
    39→    /(?:^\d+\. .+\n?)+/gm,
    40→    (block) =>
    41→      '<ol>' +
    42→      block
    43→        .trim()
    44→        .split('\n')
    45→        .map((l) => `<li>${l.replace(/^\d+\. /, '').trim()}</li>`)
    46→        .join('') +
    47→      '</ol>'
    48→  )
    49→  const BLOCK_START = /^<(ul|ol|pre|h[1-6]|blockquote)/
    50→  out = out
    51→    .split(/\n{2,}/)
    52→    .map((block) => {
    53→      const trimmed = block.trim()
    54→      if (!trimmed) return ''
    55→      if (BLOCK_START.test(trimmed)) return trimmed
    56→      return `<p>${trimmed.replace(/\n/g, '<br>')}</p>`
    57→    })
    58→    .filter(Boolean)
    59→    .join('')
    60→  return out
    61→}

    63→// ─── Architecture step renderer ───
    64→function renderArchSteps(steps: { label: string; desc?: string; type?: string; inputId?: string; children?: typeof steps }[]): string {
    65→  if (!steps || !steps.length) return ''
    66→  return steps
    67→    .map((step) => {
    68→      let typeClass = `step-type-${step.type || 'process'}`
    69→      let childrenHtml = ''
    70→      if (step.children && step.children.length > 0) {
    71→        typeClass += ' step-type-container'
    72→        childrenHtml = `<div class="step-children">${renderArchSteps(step.children)}</div>`
    73→      }
    74→      return `
    75→        <div class="flow-step">
    76→          <div class="step-box ${typeClass}">
    77→            <div class="step-label">${escapeHTML(step.label)}</div>
    78→            ${step.desc ? `<div class="step-desc">${escapeHTML(step.desc)}</div>` : ''}
    79→          </div>
    80→          ${childrenHtml}
    81→        </div>
    82→      `
    83→    })
    84→    .join('')
    85→}

   116→function Toast({ toast }: { toast: ToastData }) {
   117→  const bg =
   118→    toast.type === 'success'
   119→      ? 'var(--accent-success, #10b981)'
   120→      : toast.type === 'error'
   121→        ? 'var(--accent-danger, #ef4444)'
   122→        : 'var(--accent-primary, #0891b2)'
   123→
   124→  return (
   125→    <div
   126→      key={toast.key}
   127→      className={`app-toast ${toast.exiting ? 'toast-exit' : ''}`}
   128→      style={{ background: bg }}
   129→      role="status"
   130→      aria-live="polite"
   131→    >
   132→      {toast.message}
   133→    </div>
   134→  )
   135→}

   529→    while (terms.find((t) => t.id === id)) {
   530→      id = `${baseId}-${++suffix}`
   531→    }
   532→
   533→    const related = relatedStr
   534→      .split(',')
   535→      .map((s) => s.trim().toLowerCase().replace(/\s+/g, '-').replace(/[^a-z0-9-]/g, ''))
   536→      .filter(Boolean)
   537→
   538→    const tags = tagsStr
   539→      .split(',')
   540→      .map((s) => s.trim())
   541→      .filter(Boolean)
   542→
   543→    addTerm({
   544→      id, name, fullName: name, category: categoryId, type: 'technique',
   545→      shortDesc, definition, related, tags, codeExample: '',
   546→      createdAt: new Date().toISOString(), importance: 0,
   547→    })
   548→
   549→    if (graphRef.current) {
   550→      const cats = categories.map((c) => ({ id: c.id, name: c.name, color: c.color }))
   551→      const updatedTerms = useTitanMLStore.getState().terms
   552→      const termsData = updatedTerms.map((t) => ({
   553→        id: t.id, name: t.name, fullName: t.fullName,
   554→        category: t.category, type: t.type, shortDesc: t.shortDesc,
   555→        tags: t.tags, related: t.related,
   556→      }))
   557→      graphRef.current.loadData(cats, termsData)
   558→    }
   559→
   560→    setAddModalOpen(false)
   561→    showToast(`"${name}" added successfully!`, 'success')
   562→  }, [addTerm, categories, terms, setAddModalOpen, showToast])
   563→
   564→  // Architecture handlers
   565→  const showArchGallery = useCallback(() => {
   566→    setActiveView('architecture')
   567→    setArchCurrentView('gallery')
   568→    setArchCurrentId(null)
   569→  }, [setActiveView, setArchCurrentView, setArchCurrentId])

   573→    setArchCurrentView('architecture')

   576→  const hideArchView = useCallback(() => {

   578→    setArchCurrentId(null)
   579→  }, [setActiveView, setArchCurrentId])

   596→      setActivePath(path)
   597→    }
   598→  }, [setPathMode, setPathSubPaths, setActivePath])

   606→    }
   607→  }, [terms, setActiveView, setSelectedTerm, navigateTerm])

   611→    if (!selectedTerm) return null
   612→    return categories.find((c) => c.id === selectedTerm.category) ?? null
   613→  }, [selectedTerm, categories])

   619→      const rt = terms.find((t) => t.id === selectedTerm.related[i])
   620→      if (!rt) continue
   621→      const rc = categories.find((c) => c.id === rt.category)
   622→      results.push({ term: rt, cat: rc })
   623→    }
   624→    return results
   625→  }, [selectedTerm, terms, categories])
   626→
   627→  const currentArch = useMemo(() => {
   628→    if (!archCurrentId) return null
   629→    return architectures.find((a) => a.id === archCurrentId) ?? null
   630→  }, [archCurrentId, architectures])
   631→
   632→  // ─── Loading screen ───
   633→  if (!isLoaded) {

   636→
   637→  // Show fade-out loader + app content simultaneously
