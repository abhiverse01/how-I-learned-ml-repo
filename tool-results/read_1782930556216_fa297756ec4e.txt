     8→     8→function escapeHTML(text: string): string {
     9→     9→  return String(text ?? '')
    10→    10→    .replace(/&/g, '&amp;')
    11→    11→    .replace(/</g, '&lt;')
    12→    12→    .replace(/>/g, '&gt;')
    13→    13→    .replace(/"/g, '&quot;')
    14→    14→    .replace(/'/g, '&#39;')
    15→    15→}

    17→    17→function parseMarkdown(text: string): string {
    18→    18→  if (!text) return ''
    19→    19→  let out = escapeHTML(text)
    20→    20→  out = out.replace(/```([\s\S]*?)```/g, (_, code) => `<pre><code>${code.trim()}</code></pre>`)
    21→    21→  out = out.replace(/`([^`\n]+)`/g, '<code>$1</code>')
    22→    22→  out = out.replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>')
    23→    23→  out = out.replace(/(?<!\*)\*(?!\*)(.+?)(?<!\*)\*(?!\*)/g, '<em>$1</em>')
    24→    24→  out = out.replace(/^### (.+)$/gm, '<h4>$1</h4>')
    25→    25→  out = out.replace(/^## (.+)$/gm, '<h3>$1</h3>')
    26→    26→  out = out.replace(/^# (.+)$/gm, '<h2>$1</h2>')
    27→    27→  out = out.replace(
    28→    28→    /(?:^- .+\n?)+/gm,
    29→    29→    (block) =>
    30→    30→      '<ul>' +
    31→    31→      block
    32→    32→        .trim()
    33→    33→        .split('\n')
    34→    34→        .map((l) => `<li>${l.replace(/^- /, '').trim()}</li>`)
    35→    35→        .join('') +
    36→    36→      '</ul>'
    37→    37→  )
    38→    38→  out = out.replace(
    39→    39→    /(?:^\d+\. .+\n?)+/gm,
    40→    40→    (block) =>
    41→    41→      '<ol>' +
    42→    42→      block
    43→    43→        .trim()
    44→    44→        .split('\n')
    45→    45→        .map((l) => `<li>${l.replace(/^\d+\. /, '').trim()}</li>`)
    46→    46→        .join('') +
    47→    47→      '</ol>'
    48→    48→  )
    49→    49→  const BLOCK_START = /^<(ul|ol|pre|h[1-6]|blockquote)/
    50→    50→  out = out
    51→    51→    .split(/\n{2,}/)
    52→    52→    .map((block) => {
    53→    53→      const trimmed = block.trim()
    54→    54→      if (!trimmed) return ''
    55→    55→      if (BLOCK_START.test(trimmed)) return trimmed
    56→    56→      return `<p>${trimmed.replace(/\n/g, '<br>')}</p>`
    57→    57→    })
    58→    58→    .filter(Boolean)
    59→    59→    .join('')
    60→    60→  return out
    61→    61→}

    63→    63→// ─── Architecture step renderer ───
    64→    64→function renderArchSteps(steps: { label: string; desc?: string; type?: string; inputId?: string; children?: typeof steps }[]): string {
    65→    65→  if (!steps || !steps.length) return ''
    66→    66→  return steps
    67→    67→    .map((step) => {
    68→    68→      let typeClass = `step-type-${step.type || 'process'}`
    69→    69→      let childrenHtml = ''
    70→    70→      if (step.children && step.children.length > 0) {
    71→    71→        typeClass += ' step-type-container'
    72→    72→        childrenHtml = `<div class="step-children">${renderArchSteps(step.children)}</div>`
    73→    73→      }
    74→    74→      return `
    75→    75→        <div class="flow-step">
    76→    76→          <div class="step-box ${typeClass}">
    77→    77→            <div class="step-label">${escapeHTML(step.label)}</div>
    78→    78→            ${step.desc ? `<div class="step-desc">${escapeHTML(step.desc)}</div>` : ''}
    79→    79→          </div>
    80→    80→          ${childrenHtml}
    81→    81→        </div>
    82→    82→      `
    83→    83→    })
    84→    84→    .join('')
    85→    85→}

   126→   126→      key={toast.key}
   127→   127→      className={`app-toast ${toast.exiting ? 'toast-exit' : ''}`}
   128→   128→      style={{ background: bg }}

   130→   130→      aria-live="polite"

   439→   439→        ta.value = code
   440→   440→        ta.style.cssText = 'position:fixed;top:-999px;left:-999px;opacity:0;'
   441→   441→        document.body.appendChild(ta)
   442→   442→        ta.select()
   443→   443→        document.execCommand('copy')
   444→   444→        ta.remove()
   445→   445→      }
   446→   446→      setCopyState('copied')
   447→   447→      setTimeout(() => setCopyState('idle'), 1500)
   448→   448→    } catch {
   449→   449→      setCopyState('failed')
   450→   450→      setTimeout(() => setCopyState('idle'), 1500)
   451→   451→    }
   452→   452→  }, [selectedTerm?.codeExample])

   457→   457→  }, [selectedTerm])

   465→   465→      setSelectedCategory(categoryId)
   466→   466→      graphRef.current?.filterByCategory(categoryId)

   468→   468→  }, [selectedCategory, setSelectedCategory])

   513→   513→
   514→   514→    const related = relatedStr
   515→   515→      .split(',')
   516→   516→      .map((s) => s.trim().toLowerCase().replace(/\s+/g, '-').replace(/[^a-z0-9-]/g, ''))
   517→   517→      .filter(Boolean)
   518→   518→
   519→   519→    const tags = tagsStr
   520→   520→      .split(',')
   521→   521→      .map((s) => s.trim())
   522→   522→      .filter(Boolean)
   523→   523→
   524→   524→    addTerm({
   525→   525→      id, name, fullName: name, category: categoryId, type: 'technique',
   526→   526→      shortDesc, definition, related, tags, codeExample: '',
   527→   527→      createdAt: new Date().toISOString(), importance: 0,
   528→   528→    })
   529→   529→
   530→   530→    if (graphRef.current) {

   534→   534→        id: t.id, name: t.name, fullName: t.fullName,

   536→   536→        tags: t.tags, related: t.related,
   537→   537→      }))
   538→   538→      graphRef.current.loadData(cats, termsData)
   539→   539→    }
   540→   540→
   541→   541→    setAddModalOpen(false)
   542→   542→    showToast(`"${name}" added successfully!`, 'success')
   543→   543→  }, [addTerm, categories, terms, setAddModalOpen, showToast])
   544→   544→
   545→   545→  // Architecture handlers
   546→   546→  const showArchGallery = useCallback(() => {

   559→   559→    setArchCurrentId(null)

   566→   566→    setActivePath(null)
   567→   567→    setPathSubPaths([])
   568→   568→  }, [setActiveView, setPathMode, setActivePath, setPathSubPaths])

   572→   572→      setPathMode('subpaths')
   573→   573→      setPathSubPaths(path.paths)
   574→   574→      setActivePath(path)
   575→   575→    } else {
   576→   576→      setPathMode('steps')
   577→   577→      setActivePath(path)
   578→   578→    }
   579→   579→  }, [setPathMode, setPathSubPaths, setActivePath])
   580→   580→
   581→   581→  const focusTerm = useCallback((termName: string) => {
   582→   582→    setActiveView('graph')
   583→   583→    const term = terms.find((t) => t.name === termName || t.fullName === termName)
   584→   584→    if (term) {
   585→   585→      setSelectedTerm(term)
   586→   586→      navigateTerm(term.id)
   587→   587→    }
   588→   588→  }, [terms, setActiveView, setSelectedTerm, navigateTerm])
   589→   589→
   590→   590→  // ─── Memoized derived data ───
   591→   591→  const selectedTermCategory = useMemo(() => {
   592→   592→    if (!selectedTerm) return null
   593→   593→    return categories.find((c) => c.id === selectedTerm.category) ?? null
   594→   594→  }, [selectedTerm, categories])
   595→   595→
   596→   596→  const relatedTerms = useMemo(() => {

   600→   600→      const rt = terms.find((t) => t.id === selectedTerm.related[i])
   601→   601→      if (!rt) continue
   602→   602→      const rc = categories.find((c) => c.id === rt.category)
   603→   603→      results.push({ term: rt, cat: rc })
   604→   604→    }
   605→   605→    return results
   606→   606→  }, [selectedTerm, terms, categories])
   607→   607→
   608→   608→  const currentArch = useMemo(() => {
   609→   609→    if (!archCurrentId) return null
   610→   610→    return architectures.find((a) => a.id === archCurrentId) ?? null
   611→   611→  }, [archCurrentId, architectures])
   612→   612→
   613→   613→  // ─── Loading screen ───
   614→   614→  if (!isLoaded) {
   615→   615→    return <LoadingScreen />
   616→   616→  }
   617→   617→
   618→   618→  // Show fade-out loader + app content simultaneously
   619→   619→  return (
   620→   620→    <>
   621→   621→      {loaderFading && (
   622→   622→        <div id="initial-loader" className="fade-out">
   623→   623→          <div className="loader-spinner" />
   624→   624→          <p className="loader-text">
   625→   625→            Initializing Knowledge Base...
   626→   626→          </p>
   627→   627→        </div>
   628→   628→      )}
   629→   629→      <AppContent

   633→   633→          sidebarOpen={sidebarOpen} sidebarCollapsed={sidebarCollapsed} closeSidebar={closeSidebar}
   634→   634→          categories={categories} selectedCategory={selectedCategory} stats={stats}
   635→   635→          selectedFilter={selectedFilter} isDarkMode={isDarkMode} toggleDarkMode={toggleDarkMode}
   636→   636→          handleCategoryClick={handleCategoryClick} handleFilterClick={handleFilterClick}
   637→   637→          showArchGallery={showArchGallery} showPathsView={showPathsView}

   651→   651→          terms={terms}
   652→   652→          // Detail panel
   653→   653→          detailPanelOpen={detailPanelOpen} selectedTerm={selectedTerm}
   654→   654→          selectedTermCategory={selectedTermCategory} relatedTerms={relatedTerms}
   655→   655→          copyState={copyState}
   656→   656→          onCopyCode={handleCopyCode}
   657→   657→          onRelatedClick={handleRelatedClick} onClosePanel={() => setDetailPanelOpen(false)}
